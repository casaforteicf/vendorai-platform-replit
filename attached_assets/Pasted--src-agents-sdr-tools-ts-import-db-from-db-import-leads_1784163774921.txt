// src/agents/sdr-tools.ts
import { db } from '../db';
import { leads, conversations, messages, tasks } from '../schema'; // Ajuste para suas tabelas reais
import { eq, desc } from 'drizzle-orm';
import { sendWhatsApp } from '../integrations/zapi'; // <-- Importando ZAPI

// 1. Buscar contexto do lead (usa conversations + messages)
export const getLeadContext = async (leadId: string) => {
  const leadResult = await db.select().from(leads).where(eq(leads.id, leadId));
  const lead = leadResult[0];
  if (!lead) throw new Error('Lead não encontrado');

  // Busca conversa ativa
  const convResult = await db.select()
    .from(conversations)
    .where(eq(conversations.lead_id, leadId))
    .limit(1);
  
  let conversation = convResult[0];
  if (!conversation) {
    const [newConv] = await db.insert(conversations)
      .values({ lead_id: leadId, status: 'active' })
      .returning();
    conversation = newConv;
  }

  // Busca histórico de mensagens da conversa
  const history = await db.select()
    .from(messages)
    .where(eq(messages.conversation_id, conversation.id))
    .orderBy(desc(messages.created_at))
    .limit(10);

  return { lead, conversation, history: history.reverse() };
};

// 2. Enviar mensagem (usa ZAPI)
export const sendMessage = async (leadId: string, text: string) => {
  const lead = await db.select().from(leads).where(eq(leads.id, leadId)).then(r => r[0]);
  if (!lead) throw new Error('Lead não encontrado');

  // Envia via WhatsApp (ZAPI)
  await sendWhatsApp(lead.phone, text);

  // Busca conversa ativa
  let conv = await db.select().from(conversations).where(eq(conversations.lead_id, leadId)).limit(1);
  let convId: string;
  if (conv.length === 0) {
    const [newConv] = await db.insert(conversations).values({ lead_id: leadId, status: 'active' }).returning();
    convId = newConv.id;
  } else {
    convId = conv[0].id;
  }

  // Registra mensagem enviada
  await db.insert(messages).values({
    conversation_id: convId,
    sender: 'agent',
    text: text,
    created_at: new Date()
  });

  return { success: true };
};

// 3. Atualizar etapa do lead
export const updateStage = async (leadId: string, stage: string) => {
  await db.update(leads)
    .set({ stage: stage, updated_at: new Date() })
    .where(eq(leads.id, leadId));
  return { success: true, new_stage: stage };
};

// 4. Agendar follow-up
export const scheduleFollowUp = async (leadId: string, dateTime: string, notes: string) => {
  await db.insert(tasks).values({
    lead_id: leadId,
    due_date: new Date(dateTime),
    notes: notes,
    status: 'pending',
    created_at: new Date()
  });
  return { success: true, scheduled_for: dateTime };
};
