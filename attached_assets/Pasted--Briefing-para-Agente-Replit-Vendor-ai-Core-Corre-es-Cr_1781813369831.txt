## PRIORIDADE 2 — Completar Webhook Instagram (ALTO IMPACTO)

### Problema
O endpoint `POST /api/webhooks/meta/instagram` recebe mensagens do Instagram
mas é um **esqueleto vazio** — não cria lead, não cria conversa, não processa
a mensagem. Clientes que mandam DM no Instagram são ignorados pelo sistema.

### Objetivo
Replicar exatamente o mesmo fluxo do webhook WhatsApp, adaptado para Instagram.

### O que fazer

**Passo 1 — Localizar o webhook atual:**
```
Arquivo: provavelmente server/routes/webhooks.ts ou server/routes/meta.ts
Buscar por: /webhooks/meta/instagram ou instagram_messaging
```

**Passo 2 — Entender o payload do Instagram (Meta Graph API)**

Payload que chega do Instagram quando cliente manda DM:
```json
{
  "object": "instagram",
  "entry": [{
    "id": "PAGE_ID",
    "time": 1234567890,
    "messaging": [{
      "sender": { "id": "INSTAGRAM_SCOPED_USER_ID" },
      "recipient": { "id": "PAGE_ID" },
      "timestamp": 1234567890,
      "message": {
        "mid": "MESSAGE_ID",
        "text": "Olá, quero saber sobre seus produtos"
      }
    }]
  }]
}
```

Para áudio/mídia:
```json
{
  "message": {
    "mid": "MESSAGE_ID",
    "attachments": [{
      "type": "audio",
      "payload": { "url": "https://..." }
    }]
  }
}
```

**Passo 3 — Implementar o handler completo**

Substituir o esqueleto pelo handler completo:

```typescript
// POST /api/webhooks/meta/instagram
async function handleInstagramWebhook(req: Request, res: Response) {
  // Responder imediatamente para o Meta não reenviar
  res.status(200).send('EVENT_RECEIVED');

  try {
    const body = req.body;
    if (body.object !== 'instagram') return;

    for (const entry of body.entry || []) {
      for (const messagingEvent of entry.messaging || []) {
        
        const instagramScopedId = messagingEvent.sender?.id;
        const recipientId = messagingEvent.recipient?.id; // PAGE_ID do tenant
        const timestamp = messagingEvent.timestamp;
        
        if (!instagramScopedId || !messagingEvent.message) continue;
        
        // Ignorar mensagens enviadas pelo próprio bot (echo)
        if (messagingEvent.message?.is_echo) continue;

        // Descobrir qual tenant é dono desta page
        const tenant = await db.query.agent_configs.findFirst({
          where: eq(agentConfigs.instagramPageId, recipientId)
        });
        
        // Fallback: buscar pelo instagram_access_token que contém este page
        // (implementar lógica de lookup por tenant)
        if (!tenant) {
          console.warn(`Tenant não encontrado para Instagram page: ${recipientId}`);
          continue;
        }

        const tenantId = tenant.tenantId;

        // Extrair texto ou transcrever áudio
        let textoMensagem = '';
        let tipoConteudo = 'text';

        if (messagingEvent.message.text) {
          textoMensagem = messagingEvent.message.text;
          tipoConteudo = 'text';
        } else if (messagingEvent.message.attachments) {
          const attachment = messagingEvent.message.attachments[0];
          
          if (attachment.type === 'audio') {
            // Transcrever com Whisper (mesmo serviço do WhatsApp)
            textoMensagem = await transcribeAudio(attachment.payload.url);
            tipoConteudo = 'audio';
          } else if (attachment.type === 'image') {
            // Interpretar com GPT-4o Vision (mesmo serviço do WhatsApp)
            textoMensagem = await interpretMedia(attachment.payload.url, 'image');
            tipoConteudo = 'image';
          } else {
            textoMensagem = `[${attachment.type} recebido]`;
            tipoConteudo = attachment.type;
          }
        }

        if (!textoMensagem) continue;

        // Buscar ou criar lead pelo instagram_scoped_id
        let lead = await db.query.leads.findFirst({
          where: and(
            eq(leads.tenantId, tenantId),
            eq(leads.instagramScopedId, instagramScopedId)
          )
        });

        if (!lead) {
          // Criar novo lead
          const [novoLead] = await db.insert(leads).values({
            tenantId,
            nome: `Lead Instagram ${instagramScopedId.slice(-6)}`,
            instagramScopedId,
            source: 'instagram',
            status: 'novo',
            canal_preferido: 'instagram',
            criadoEm: new Date()
          }).returning();
          lead = novoLead;
        }

        // Buscar ou criar conversa
        let conversa = await db.query.conversations.findFirst({
          where: and(
            eq(conversations.tenantId, tenantId),
            eq(conversations.leadId, lead.id),
            eq(conversations.channel, 'instagram'),
            eq(conversations.status, 'aberta')
          )
        });

        if (!conversa) {
          const [novaConversa] = await db.insert(conversations).values({
            tenantId,
            leadId: lead.id,
            channel: 'instagram',
            status: 'aberta',
            ultimaMensagemEm: new Date()
          }).returning();
          conversa = novaConversa;
        }

        // Inserir mensagem
        await db.insert(messages).values({
          conversationId: conversa.id,
          leadId: lead.id,
          tenantId,
          direcao: 'inbound',
          canal: 'instagram',
          conteudo: textoMensagem,
          tipoConteudo,
          enviadoPor: 'lead',
          criadoEm: new Date(timestamp)
        });

        // Atualizar lead e conversa
        await db.update(leads)
          .set({ ultimaAtividadeEm: new Date() })
          .where(eq(leads.id, lead.id));

        await db.update(conversations)
          .set({ 
            ultimaMensagemEm: new Date(),
            naoLidas: sql`${conversations.naoLidas} + 1`
          })
          .where(eq(conversations.id, conversa.id));

        // Disparar agente IA em background (igual ao WhatsApp)
        dispatchAgentBackground({
          tenantId,
          leadId: lead.id,
          conversationId: conversa.id,
          mensagem: textoMensagem,
          canal: 'instagram'
        }).catch(err => console.error('Agent dispatch error:', err));
      }
    }
  } catch (error) {
    console.error('Erro no webhook Instagram:', error);
  }
}
```

export async function generateSiteHTML(params: SiteParams): Promise<string> {

    return await generateWithClaude(params);
  }
  return await generateWithOpenAI(params);

async function generateWithClaude(params: SiteParams): Promise<string> {
  const client = new Anthropic({ apiKey: process.env.ANTHROPIC_API_KEY });
  
  const message = await client.messages.create({
    model: 'claude-sonnet-4-6',  // SEMPRE este modelo
    max_tokens: 8192,
    messages: [{
      role: 'user',
      content: buildSitePrompt(params)  // prompt detalhado existente
    }]
  });
  
  const text = message.content[0].type === 'text' ? message.content[0].text : '';
  return extractHTML(text);
}
