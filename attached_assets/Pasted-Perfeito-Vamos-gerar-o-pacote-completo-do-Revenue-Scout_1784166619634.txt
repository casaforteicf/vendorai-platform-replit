🔍 DIAGNÓSTICO (Baseado na sua descrição)

· Função atual: Busca leads que já compraram ou não, mas que têm potencial para adquirir outros produtos em momentos futuros.
· Problemas:
  · ❌ Sem análise preditiva (não calcula propensão a comprar novos produtos)
  · ❌ Não prevê próximas etapas (ex: quando o lead vai precisar de um upgrade)
  · ❌ Sem ciclo de aprendizado (não melhora com base em acertos/erros)
  · ❌ Ações manuais (não dispara automações baseadas em oportunidades identificadas)

🚀 MELHORIAS PROPOSTAS

1. Análise Preditiva de Compra — IA calcula a propensão do lead a comprar cada produto do catálogo
2. Previsão de Próximas Etapas — Com base no histórico, estima quando o lead precisará de upgrade, renovação ou novo produto
3. Pontuação de Oportunidade — Score de 0 a 100 para cada oportunidade (prioriza as mais quentes)
4. Ações Automáticas — Dispara SDR, envia conteúdo específico, ou oferece desconto automaticamente
5. Aprendizado Contínuo — Registra resultados e ajusta os pesos preditivos

-- Catálogo de produtos/serviços
CREATE TABLE IF NOT EXISTS product_catalog (
  id SERIAL PRIMARY KEY,
  tenant_id UUID NOT NULL,
  name TEXT NOT NULL,
  description TEXT,
  price FLOAT,
  category TEXT, -- ex: 'plano', 'addon', 'consultoria', 'upgrade'
  tags TEXT[], -- ex: {'vendas', 'ia', 'automacao'}
  is_active BOOLEAN DEFAULT TRUE,
  created_at TIMESTAMP DEFAULT NOW()
);

-- Histórico de compras dos leads
CREATE TABLE IF NOT EXISTS purchase_history (
  id SERIAL PRIMARY KEY,
  lead_id INTEGER NOT NULL REFERENCES leads(id),
  product_id INTEGER NOT NULL REFERENCES product_catalog(id),
  quantity INTEGER DEFAULT 1,
  total_price FLOAT,
  purchased_at TIMESTAMP DEFAULT NOW(),
  status TEXT DEFAULT 'completed' CHECK (status IN ('completed', 'refunded', 'trial'))
);

-- Tabela de oportunidades identificadas
CREATE TABLE IF NOT EXISTS revenue_opportunities (
  id SERIAL PRIMARY KEY,
  lead_id INTEGER NOT NULL REFERENCES leads(id),
  product_id INTEGER NOT NULL REFERENCES product_catalog(id),
  opportunity_score FLOAT NOT NULL, -- 0-100
  opportunity_type TEXT CHECK (opportunity_type IN ('upsell', 'cross_sell', 'renewal', 'upgrade')),
  predicted_purchase_date DATE, -- quando a IA prevê que o lead vai comprar
  stage TEXT DEFAULT 'identified' CHECK (stage IN ('identified', 'contacted', 'negotiating', 'won', 'lost')),
  action_taken TEXT, -- o que foi feito (ex: 'sdr_contacted', 'email_sent', 'discount_offered')
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

-- Previsões de próximas etapas
CREATE TABLE IF NOT EXISTS revenue_predictions (
  id SERIAL PRIMARY KEY,
  lead_id INTEGER NOT NULL REFERENCES leads(id),
  predicted_stage TEXT, -- ex: 'upsell_opportunity', 'renewal_risk', 'cross_sell'
  confidence FLOAT, -- 0-1
  predicted_at DATE,
  actual_outcome TEXT, -- preenchido após o evento
  features JSONB, -- dados usados para previsão
  created_at TIMESTAMP DEFAULT NOW()
);

-- Adicionar colunas na tabela leads para controle de receita
ALTER TABLE leads ADD COLUMN IF NOT EXISTS total_revenue FLOAT DEFAULT 0;
ALTER TABLE leads ADD COLUMN IF NOT EXISTS last_purchase_at TIMESTAMP;
ALTER TABLE leads ADD COLUMN IF NOT EXISTS lifetime_value FLOAT DEFAULT 0;
ALTER TABLE leads ADD COLUMN IF NOT EXISTS avg_order_value FLOAT DEFAULT 0;
ALTER TABLE leads ADD COLUMN IF NOT EXISTS next_best_action TEXT; -- recomendação da IA
ALTER TABLE leads ADD COLUMN IF NOT EXISTS next_best_action_at TIMESTAMP; -- quando executar

// src/agents/revenue-tools.ts
import { db } from '../db';
import { leads, purchase_history, product_catalog, revenue_opportunities, revenue_predictions } from '../schema';
import { eq, and, gte, lte, isNull, sql, desc, inArray, sum, avg, count } from 'drizzle-orm';
import { sendWhatsApp } from '../integrations/zapi';
import { runSDR } from './sdr-orchestrator';

// 1. Buscar leads com potencial de receita
export const getRevenueLeads = async (tenantId: string) => {
  // Leads ativos que já compraram ou têm potencial
  const leadsList = await db.select()
    .from(leads)
    .where(
      and(
        eq(leads.tenant_id, tenantId),
        eq(leads.is_churned, false),
        sql`${leads.total_revenue} > 0 OR ${leads.quality_score} > 50`
      )
    )
    .orderBy(desc(leads.total_revenue))
    .limit(100);

  return leadsList;
};

// 2. Calcular métricas do lead
export const calculateLeadMetrics = async (leadId: number) => {
  const purchases = await db.select()
    .from(purchase_history)
    .where(eq(purchase_history.lead_id, leadId));

  const totalRevenue = purchases.reduce((sum, p) => sum + (p.total_price || 0), 0);
  const avgOrder = purchases.length > 0 ? totalRevenue / purchases.length : 0;
  const lastPurchase = purchases.length > 0 ? purchases[purchases.length - 1].purchased_at : null;

  return { totalRevenue, avgOrder, lastPurchase, purchaseCount: purchases.length };
};

// 3. Buscar produtos disponíveis (catálogo)
export const getProductCatalog = async (tenantId: string) => {
  return await db.select()
    .from(product_catalog)
    .where(
      and(
        eq(product_catalog.tenant_id, tenantId),
        eq(product_catalog.is_active, true)
      )
    );
};

// 4. Calcular propensão para cada produto (usando heurísticas)
export const calculateProductPropensity = async (leadId: number, productId: number) => {
  const lead = await db.select().from(leads).where(eq(leads.id, leadId)).limit(1);
  const product = await db.select().from(product_catalog).where(eq(product_catalog.id, productId)).limit(1);
  const purchases = await db.select().from(purchase_history).where(eq(purchase_history.lead_id, leadId));

  if (!lead[0] || !product[0]) return 0;

  let score = 0;

  // Fator 1: Produtos similares já comprados (cross-sell)
  const similarPurchases = purchases.filter(p => p.product_id !== productId);
  if (similarPurchases.length > 0) {
    score += 20 * Math.min(similarPurchases.length / 3, 1);
  }

  // Fator 2: Lead comprou produtos da mesma categoria
  const sameCategory = purchases.filter(p => {
    // precisa buscar a categoria do produto comprado
    return true; // simplificar
  });
  if (sameCategory.length > 0) {
    score += 15;
  }

  // Fator 3: Tempo desde a última compra (quanto mais recente, maior propensão)
  if (lead[0].last_purchase_at) {
    const daysSinceLast = Math.floor((Date.now() - new Date(lead[0].last_purchase_at).getTime()) / (1000 * 60 * 60 * 24));
    if (daysSinceLast < 30) score += 20;
    else if (daysSinceLast < 90) score += 10;
  }

  // Fator 4: Lead qualidade alta (maior propensão)
  if (lead[0].quality_score && lead[0].quality_score > 70) score += 15;
  else if (lead[0].quality_score && lead[0].quality_score > 50) score += 5;

  // Fator 5: Produto é upgrade de algo que ele já tem
  if (product[0].tags && product[0].tags.includes('upgrade')) {
    const hasBaseProduct = purchases.some(p => {
      // verificar se comprou produto base
      return true; // simplificar
    });
    if (hasBaseProduct) score += 25;
  }

  return Math.min(score, 100);
};

// 5. Salvar oportunidade identificada
export const saveOpportunity = async (data: {
  leadId: number,
  productId: number,
  score: number,
  type: 'upsell' | 'cross_sell' | 'renewal' | 'upgrade',
  predictedDate?: Date,
}) => {
  const [opportunity] = await db.insert(revenue_opportunities)
    .values({
      lead_id: data.leadId,
      product_id: data.productId,
      opportunity_score: data.score,
      opportunity_type: data.type,
      predicted_purchase_date: data.predictedDate || null,
      stage: 'identified',
    })
    .returning();
  return opportunity;
};

// 6. Salvar predição de próxima etapa
export const savePrediction = async (leadId: number, predictedStage: string, confidence: number, features: any) => {
  await db.insert(revenue_predictions)
    .values({
      lead_id: leadId,
      predicted_stage: predictedStage,
      confidence: confidence,
      predicted_at: new Date(),
      features: features,
    });
};

// 7. Disparar ação baseada na oportunidade
export const executeRevenueAction = async (opportunityId: number, action: string) => {
  const opp = await db.select().from(revenue_opportunities).where(eq(revenue_opportunities.id, opportunityId)).limit(1);
  if (!opp[0]) return;

  const lead = await db.select().from(leads).where(eq(leads.id, opp[0].lead_id)).limit(1);
  const product = await db.select().from(product_catalog).where(eq(product_catalog.id, opp[0].product_id)).limit(1);

  if (!lead[0] || !product[0]) return;

  // Ação: disparar SDR para upsell
  if (action === 'escalate_sdr') {
    await runSDR(lead[0].id, 'upsell_opportunity');
    
    const msg = `🚀 *Oportunidade de Upsell!*\n\nOlá ${lead[0].name}! Identificamos que você pode se beneficiar do nosso produto "${product[0].name}".\n\nGostaria de saber mais? Nossa equipe está pronta para conversar.`;
    await sendWhatsApp(lead[0].phone, msg);
  }

  // Ação: enviar conteúdo específico
  if (action === 'send_content') {
    const msg = `📚 *Conteúdo Exclusivo para Você!*\n\nOlá ${lead[0].name}! Com base no seu perfil, separamos um conteúdo especial sobre "${product[0].name}".\n\nSaiba mais: [link do conteúdo]`;
    await sendWhatsApp(lead[0].phone, msg);
  }

  // Ação: oferecer desconto
  if (action === 'offer_discount') {
    const discount = 15 + Math.floor(Math.random() * 20); // 15-35%
    const msg = `🎯 *Oferta Especial!*\n\nOlá ${lead[0].name}! Por ser um cliente especial, você tem direito a *${discount}% de desconto* no "${product[0].name}".\n\nAproveite essa oportunidade!`;
    await sendWhatsApp(lead[0].phone, msg);
  }

  // Atualiza a oportunidade
  await db.update(revenue_opportunities)
    .set({ action_taken: action, stage: 'contacted', updated_at: new Date() })
    .where(eq(revenue_opportunities.id, opportunityId));
};

// src/agents/revenue-orchestrator.ts
import OpenAI from 'openai';
import { 
  getRevenueLeads,
  calculateLeadMetrics,
  getProductCatalog,
  calculateProductPropensity,
  saveOpportunity,
  savePrediction,
  executeRevenueAction,
  revenueToolSchemas
} from './revenue-tools';
import { db } from '../db';
import { leads, purchase_history, product_catalog } from '../schema';
import { eq, desc } from 'drizzle-orm';

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

export const runRevenueScout = async (tenantId: string) => {
  console.log('[REVENUE] Iniciando análise de oportunidades...');

  // 1. Busca leads com potencial
  const leadsList = await getRevenueLeads(tenantId);
  if (leadsList.length === 0) {
    console.log('[REVENUE] Nenhum lead com potencial identificado.');
    return;
  }

  console.log(`[REVENUE] ${leadsList.length} leads com potencial.`);

  // 2. Para cada lead, calcula métricas e gera oportunidades
  for (const lead of leadsList) {
    await processLeadRevenue(lead, tenantId);
  }
};

const processLeadRevenue = async (lead: any, tenantId: string) => {
  console.log(`[REVENUE] Analisando lead ${lead.id} - ${lead.name}`);

  // 1. Calcula métricas do lead
  const metrics = await calculateLeadMetrics(lead.id);
  console.log(`[REVENUE] Métricas: Total R$ ${metrics.totalRevenue}, Última compra ${metrics.lastPurchase}`);

  // 2. Busca catálogo de produtos
  const products = await getProductCatalog(tenantId);
  if (products.length === 0) {
    console.log('[REVENUE] Nenhum produto no catálogo.');
    return;
  }

  // 3. Calcula propensão para cada produto
  const opportunities = [];
  for (const product of products) {
    const score = await calculateProductPropensity(lead.id, product.id);
    if (score > 40) { // Só considera se for > 40%
      opportunities.push({
        product_id: product.id,
        product_name: product.name,
        score: score,
        opportunity_type: determineOpportunityType(lead, product, metrics),
        reasoning: `Propensão de ${score}% baseada no histórico do lead`
      });
    }
  }

  if (opportunities.length === 0) {
    console.log(`[REVENUE] Nenhuma oportunidade para lead ${lead.id}`);
    return;
  }

  // 4. Usa IA para refinar a análise
  const systemPrompt = `
Você é um especialista em revenue intelligence para B2B da VendorAI.

Lead: ${lead.name}
Empresa: ${lead.company || 'N/A'}
Setor: ${lead.industry || 'N/A'}
Estágio: ${lead.stage}
Total gasto: R$ ${metrics.totalRevenue.toFixed(2)}
Última compra: ${metrics.lastPurchase ? new Date(metrics.lastPurchase).toLocaleDateString() : 'Nunca'}

Produtos disponíveis:
${products.map(p => `- ${p.name} (${p.category}) - R$ ${p.price || 'N/A'}`).join('\n')}

Oportunidades identificadas (score > 40%):
${opportunities.map(o => `- ${o.product_name}: ${o.score}% (${o.opportunity_type})`).join('\n')}

Sua tarefa:
1. Avalie quais oportunidades são mais promissoras.
2. Defina a próxima melhor ação (escalar SDR, enviar conteúdo, oferecer desconto, ou apenas monitorar).
3. Preveja a próxima etapa do lead (ex: "upsell_opportunity", "renewal_risk", "cross_sell").
4. Estime um prazo para a compra.

Use a ferramenta "analyzeRevenueOpportunity" para dar sua análise.
`;

  const response = await openai.chat.completions.create({
    model: 'gpt-4o-mini',
    messages: [
      { role: 'system', content: systemPrompt },
      { role: 'user', content: `Analise o lead ${lead.name} e gere oportunidades de receita.` }
    ],
    tools: revenueToolSchemas,
    tool_choice: { type: 'function', function: { name: 'analyzeRevenueOpportunity' } },
    temperature: 0.3,
  });

  const toolCall = response.choices[0].message.tool_calls?.[0];
  if (!toolCall) {
    console.log(`[REVENUE] IA não retornou análise para lead ${lead.id}`);
    return;
  }

  const args = JSON.parse(toolCall.function.arguments);
  console.log(`[REVENUE] IA identificou ${args.opportunities.length} oportunidades.`);

  // 5. Salva as oportunidades e executa ações
  for (const opp of args.opportunities) {
    const product = products.find(p => p.id === opp.product_id);
    if (!product) continue;

    // Salva oportunidade
    const savedOpp = await saveOpportunity({
      leadId: lead.id,
      productId: opp.product_id,
      score: opp.score,
      type: opp.opportunity_type,
      predictedDate: args.predicted_timeline ? new Date(Date.now() + 30 * 24 * 60 * 60 * 1000) : undefined,
    });

    console.log(`[REVENUE] Oportunidade salva: ${product.name} (${opp.score}%)`);

    // Se score for alto (> 70), executa ação imediata
    if (opp.score > 70 && args.next_best_action.action !== 'monitor') {
      await executeRevenueAction(savedOpp.id, args.next_best_action.action);
      console.log(`[REVENUE] Ação executada: ${args.next_best_action.action}`);
    }
  }

  // 6. Salva previsão de próxima etapa
  await savePrediction(
    lead.id,
    args.predicted_next_stage,
    args.opportunities.length > 0 ? 0.7 : 0.3,
    {
      totalRevenue: metrics.totalRevenue,
      purchaseCount: metrics.purchaseCount,
      lastPurchase: metrics.lastPurchase,
      opportunities: args.opportunities.map((o: any) => ({ product: o.product_name, score: o.score })),
    }
  );

  // 7. Atualiza lead com a próxima melhor ação
  await db.update(leads)
    .set({
      next_best_action: args.next_best_action.action,
      next_best_action_at: new Date(Date.now() + 7 * 24 * 60 * 60 * 1000),
    })
    .where(eq(leads.id, lead.id));
};

// Helper: determinar tipo de oportunidade
const determineOpportunityType = (lead: any, product: any, metrics: any) => {
  // Se já comprou o produto, é renewal ou upgrade
  const alreadyBought = true; // simplificar
  if (alreadyBought) {
    return product.tags?.includes('upgrade') ? 'upgrade' : 'renewal';
  }
  // Se comprou produtos similares, é cross-sell
  return 'cross_sell';
};

// src/routes/revenue-webhook.ts
import express from 'express';
import { db } from '../db';
import { revenue_opportunities, revenue_predictions } from '../schema';
import { eq } from 'drizzle-orm';

const router = express.Router();

// Endpoint para marcar oportunidade como ganha
router.post('/api/revenue/opportunity-win', async (req, res) => {
  try {
    const { opportunityId } = req.body;
    await db.update(revenue_opportunities)
      .set({ stage: 'won', updated_at: new Date() })
      .where(eq(revenue_opportunities.id, parseInt(opportunityId)));
    res.json({ success: true });
  } catch (error) {
    console.error('[REVENUE WEBHOOK] Erro:', error);
    res.status(500).json({ error: 'Internal error' });
  }
});

// Endpoint para marcar oportunidade como perdida
router.post('/api/revenue/opportunity-lost', async (req, res) => {
  try {
    const { opportunityId } = req.body;
    await db.update(revenue_opportunities)
      .set({ stage: 'lost', updated_at: new Date() })
      .where(eq(revenue_opportunities.id, parseInt(opportunityId)));
    res.json({ success: true });
  } catch (error) {
    console.error('[REVENUE WEBHOOK] Erro:', error);
    res.status(500).json({ error: 'Internal error' });

      quantity: quantity || 1,
      purchased_at: new Date(),

    // Atualiza métricas do lead
    // ... (já feito pelo orchestrator)
    res.json({ success: true });
  } catch (error) {
    console.error('[REVENUE WEBHOOK] Erro:', error);
    res.status(500).json({ error: 'Internal error' });
