// src/agents/sdr-orchestrator.ts
import OpenAI from 'openai';
import { getLeadContext, sendMessage, updateStage, scheduleFollowUp, toolSchemas } from './sdr-tools';

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

export const runSDR = async (leadId: string, triggerEvent: 'new_lead' | 'reply_received') => {
  console.log(`[SDR] Iniciando lead ${leadId} - Evento: ${triggerEvent}`);

  const { lead, conversation, history } = await getLeadContext(leadId);
  if (!lead) return console.error('[SDR] Lead não encontrado');

  const systemPrompt = `
Você é um SDR B2B da VendorAI. Siga o fluxo BANT.
Lead: ${lead.name}, Empresa: ${lead.company || 'Não informada'}, Estágio: ${lead.stage || 'Novo'}.
Histórico (últimas 10 msgs): ${history.map(m => `${m.sender}: ${m.text}`).join('\n')}.

Regras rígidas:
- Se for "new_lead": cumprimente, apresente-se e pergunte sobre a maior dor do lead.
- Se for "reply_received": analise a resposta. Se mostrar interesse (ex: "quanto custa", "quero saber mais"), use "sendMessage" para propor reunião e "scheduleFollowUp".
- Se disser "sem dinheiro" ou "não quero", use "updateStage" para "Perdido" (não insista).
- Use as ferramentas para agir. Nunca invente dados.`;

  const response = await openai.chat.completions.create({
    model: 'gpt-4o-mini',
    messages: [
      { role: 'system', content: systemPrompt },
      { role: 'user', content: `O lead acabou de enviar: "${history[history.length-1]?.text || 'Nova conversa'}"` }
    ],
    tools: toolSchemas,
    tool_choice: 'auto',
    temperature: 0.3,
  });

  const message = response.choices[0].message;
  if (message.tool_calls) {
    for (const call of message.tool_calls) {
      const args = JSON.parse(call.function.arguments);
      console.log(`[SDR] Executando ${call.function.name}`, args);
      try {
        switch (call.function.name) {
          case 'sendMessage': await sendMessage(leadId, args.text); break;
          case 'updateStage': await updateStage(leadId, args.stage); break;
          case 'scheduleFollowUp': await scheduleFollowUp(leadId, args.dateTime, args.notes); break;
        }
      } catch (error) {
        console.error(`[SDR] Erro ao executar ${call.function.name}:`, error);
      }
    }
  }
  console.log(`[SDR] Finalizado lead ${leadId}`);
};
