## 🧠 VISÃO GERAL

O Revenue Scout agora captura sinais de intenção (pesquisas no Google, cliques em anúncios do Meta, visitas ao site, perguntas no WhatsApp) e combina com o contexto informado pelo lead para fazer predições precisas de compra.

Arquitetura:
1. Lead informa estágio (ex: "Estou levantando paredes") → Contexto real
2. Sistema captura sinais de intenção (Google, Meta, Site, WhatsApp) → Intenção real
3. Revenue Scout combina: Contexto (40%) + Comportamento (30%) + Intenção (30%) → Score preditivo
4. Ação automática baseada no score: escalar SDR, enviar conteúdo, oferecer desconto

-- 1. Tabela de Sinais de Intenção
CREATE TABLE IF NOT EXISTS intent_signals (
  id SERIAL PRIMARY KEY,
  lead_id INTEGER NOT NULL REFERENCES leads(id) ON DELETE CASCADE,
  signal_type TEXT NOT NULL CHECK (signal_type IN ('google_search', 'meta_click', 'site_visit', 'whatsapp_question')),
  source TEXT NOT NULL CHECK (source IN ('google_ads', 'instagram', 'facebook', 'site', 'whatsapp')),
  product_id INTEGER REFERENCES product_catalog(id),
  search_term TEXT,
  url TEXT,
  value FLOAT DEFAULT 50,
  confidence FLOAT DEFAULT 0.8,
  captured_at TIMESTAMP DEFAULT NOW(),
  metadata JSONB
);

CREATE INDEX idx_intent_signals_lead ON intent_signals(lead_id);
CREATE INDEX idx_intent_signals_type ON intent_signals(signal_type);
CREATE INDEX idx_intent_signals_captured ON intent_signals(captured_at DESC);

-- 2. Tabela de Matriz Preditiva (Transição de Estágios)
CREATE TABLE IF NOT EXISTS predictive_transition_matrix (
  id SERIAL PRIMARY KEY,
  tenant_id UUID NOT NULL,
  current_stage TEXT NOT NULL,
  product_id INTEGER NOT NULL REFERENCES product_catalog(id),
  avg_days_to_purchase INTEGER NOT NULL,
  base_propensity FLOAT NOT NULL CHECK (base_propensity BETWEEN 0 AND 1),
  product_category TEXT,
  confidence_interval FLOAT DEFAULT 0.8,
  sample_size INTEGER DEFAULT 0,
  created_at TIMESTAMP DEFAULT NOW(),
  updated_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_predictive_matrix_tenant_stage ON predictive_transition_matrix(tenant_id, current_stage);

-- 3. Adicionar colunas na tabela leads
ALTER TABLE leads ADD COLUMN IF NOT EXISTS construction_stage TEXT;
ALTER TABLE leads ADD COLUMN IF NOT EXISTS context_answered_at TIMESTAMP;
ALTER TABLE leads ADD COLUMN IF NOT EXISTS intent_score FLOAT DEFAULT 0;
ALTER TABLE leads ADD COLUMN IF NOT EXISTS last_intent_update TIMESTAMP;

-- 4. Tabela de Log de Predições (para aprendizado)
CREATE TABLE IF NOT EXISTS prediction_log (
  id SERIAL PRIMARY KEY,
  lead_id INTEGER NOT NULL REFERENCES leads(id),
  product_id INTEGER NOT NULL REFERENCES product_catalog(id),
  predicted_score FLOAT,
  actual_outcome TEXT CHECK (actual_outcome IN ('bought', 'not_bought', 'pending')),
  predicted_days INTEGER,
  actual_days INTEGER,
  created_at TIMESTAMP DEFAULT NOW(),
  resolved_at TIMESTAMP
);

CREATE INDEX idx_prediction_log_lead ON prediction_log(lead_id);

// src/integrations/meta-conversions.ts
import axios from 'axios';
import { db } from '../db';
import { leads } from '../schema';
import { eq } from 'drizzle-orm';

const META_PIXEL_ID = process.env.META_PIXEL_ID;
const META_ACCESS_TOKEN = process.env.META_ACCESS_TOKEN;

export const sendMetaConversion = async (leadId: number, eventName: string, productId?: number) => {
  if (!META_PIXEL_ID || !META_ACCESS_TOKEN) {
    console.warn('[META] Pixel não configurado. Pulando.');
    return;
  }

  const lead = await db.select().from(leads).where(eq(leads.id, leadId)).limit(1);
  if (!lead[0]) return;

  const payload = {
    data: [{
      event_name: eventName,
      event_time: Math.floor(Date.now() / 1000),
      user_data: {
        em: lead[0].email ? [lead[0].email] : undefined,
        ph: lead[0].phone ? [lead[0].phone.replace(/\D/g, '')] : undefined,
      },
      custom_data: {
        content_ids: productId ? [productId.toString()] : undefined,
        content_type: 'product',
      },
      action_source: 'website',
    }],
    access_token: META_ACCESS_TOKEN,
  };

  try {
    const response = await axios.post(
      `https://graph.facebook.com/v18.0/${META_PIXEL_ID}/events`,
      payload
    );
    console.log(`[META] Evento ${eventName} enviado para lead ${leadId}`);
    return response.data;
  } catch (error: any) {
    console.error('[META] Erro:', error.response?.data || error.message);
  }
};

  try {
    const { leadId, searchTerm, productId, value, source } = req.body;
    if (!leadId) return res.status(400).json({ error: 'leadId é obrigatório' });

// Mapear termo de busca para produto (regras + IA)
async function mapSearchTermToProduct(searchTerm: string) {
  const term = searchTerm.toLowerCase();
  
  // Regras simples
  const rules: Record<string, string[]> = {
    'tijolo': ['Tijolo', 'Tijolos'],
    'cimento': ['Cimento'],
    'tinta': ['Tinta', 'Tintas'],
    'argamassa': ['Argamassa'],
    'telha': ['Telha', 'Telhas'],
  };

  let productName: string | null = null;
  for (const [key, values] of Object.entries(rules)) {
    if (term.includes(key)) {
      productName = values[0];
      break;
    }
  }

  if (productName) {
    const product = await db.select()
      .from(product_catalog)
      .where(eq(product_catalog.name, productName))
      .limit(1);
    return product[0] || null;
  }
  return null;
}

// Detectar produto de uma pergunta no WhatsApp
async function detectProductFromQuestion(question: string) {
  // Usa a mesma lógica do mapSearchTermToProduct
  return mapSearchTermToProduct(question);
}

// Atualizar score de intenção do lead
async function updateLeadIntentScore(leadId: number) {
  const signals = await db.select()
    .from(intent_signals)
    .where(
      and(
        eq(intent_signals.lead_id, leadId),
        sql`${intent_signals.captured_at} > NOW() - INTERVAL '30 days'`
      )
    );

  if (signals.length === 0) {
    await db.update(leads)
      .set({ intent_score: 0, last_intent_update: new Date() })
      .where(eq(leads.id, leadId));
    return;
  }

  const weights: Record<string, number> = {
    google_search: 30,
    meta_click: 25,
    site_visit: 20,
    whatsapp_question: 40,
  };

  let total = 0;
  for (const signal of signals) {
    const weight = weights[signal.signal_type] || 10;
    total += weight * (signal.value / 100);
  }

  const score = Math.min(Math.round(total), 100);
  await db.update(leads)
    .set({ intent_score: score, last_intent_update: new Date() })
    .where(eq(leads.id, leadId));
}

// src/agents/revenue-orchestrator.ts
import { db } from '../db';
import { leads, predictive_transition_matrix, revenue_opportunities, intent_signals, product_catalog } from '../schema';
import { eq, and, isNotNull, sql, desc } from 'drizzle-orm';
import { runSDR } from './sdr-orchestrator';
import { sendWhatsApp } from '../integrations/zapi';

// ========== FUNÇÃO PRINCIPAL ==========
export const runRevenueScout = async (tenantId: string, options?: { leadId?: number, forceRecalc?: boolean }) => {
  console.log('[REVENUE] Iniciando Revenue Scout com Intenção Preditiva...');

  // 1. Busca leads para processar
  let leadsToProcess;
  if (options?.leadId) {
    leadsToProcess = await db.select().from(leads).where(eq(leads.id, options.leadId));
  } else {
    leadsToProcess = await db.select()
      .from(leads)
      .where(
        and(
          eq(leads.tenant_id, tenantId),
          isNotNull(leads.construction_stage),
          eq(leads.is_churned, false)
        )
      )
      .orderBy(desc(leads.intent_score))
      .limit(100);
  }

  if (leadsToProcess.length === 0) {
    console.log('[REVENUE] Nenhum lead com estágio definido.');
    return;
  }

  console.log(`[REVENUE] Processando ${leadsToProcess.length} leads...`);

  for (const lead of leadsToProcess) {
    await processLead(lead, tenantId);
  }
};

// ========== PROCESSAR CADA LEAD ==========
async function processLead(lead: any, tenantId: string) {
  console.log(`[REVENUE] Analisando lead ${lead.id} - ${lead.name} (Estágio: ${lead.construction_stage})`);

  // 1. Busca matriz preditiva para o estágio atual
  const predictions = await db.select()
    .from(predictive_transition_matrix)
    .where(
      and(
        eq(predictive_transition_matrix.tenant_id, tenantId),
        eq(predictive_transition_matrix.current_stage, lead.construction_stage)
      )
    );

  if (predictions.length === 0) {
    console.log(`[REVENUE] Nenhuma predição para estágio: ${lead.construction_stage}`);
    return;
  }

  // 2. Busca sinais de intenção dos últimos 30 dias
  const intentSignals = await db.select()
    .from(intent_signals)
    .where(
      and(
        eq(intent_signals.lead_id, lead.id),
        sql`${intent_signals.captured_at} > NOW() - INTERVAL '30 days'`
      )
    );

  // 3. Calcula scores
  const intentScore = calculateIntentScore(intentSignals);
  const behaviorScore = lead.quality_score || 50;
  const contextScore = 100; // Estágio confirmado pelo lead = contexto perfeito

  // 4. Atualiza score de intenção no lead
  await db.update(leads)
    .set({ intent_score: intentScore, last_intent_update: new Date() })
    .where(eq(leads.id, lead.id));

  // 5. Para cada predição, calcula score final
  for (const pred of predictions) {
    // Score baseado na matriz (confiabilidade histórica)
    const baseScore = pred.base_propensity * 100;

    // Combinação ponderada
    let finalScore = (contextScore * 0.4) + (behaviorScore * 0.2) + (intentScore * 0.4);
    
    // Ajuste específico: se o lead demonstrou intenção para este produto
    const productIntent = intentSignals.filter(s => s.product_id === pred.product_id);
    if (productIntent.length > 0) {
      const bonus = Math.min(productIntent.length * 5, 15);
      finalScore = Math.min(finalScore + bonus, 100);
    }

    // Ajuste de timing
    const daysSinceLastPurchase = lead.last_purchase_at 
      ? (Date.now() - new Date(lead.last_purchase_at).getTime()) / (1000 * 60 * 60 * 24)
      : 999;
    
    if (daysSinceLastPurchase > pred.avg_days_to_purchase) {
      finalScore = Math.min(finalScore + 10, 100); // Urgência
    }

    finalScore = Math.round(Math.max(0, Math.min(finalScore, 100)));

    console.log(`[REVENUE] Produto ${pred.product_id}: Score ${finalScore}%`);

    // 6. Salva oportunidade
    const [opportunity] = await db.insert(revenue_opportunities)
      .values({
        lead_id: lead.id,
        product_id: pred.product_id,
        opportunity_score: finalScore,
        opportunity_type: determineOpportunityType(lead, pred),
        predicted_purchase_date: new Date(Date.now() + (pred.avg_days_to_purchase * 24 * 60 * 60 * 1000)),
        stage: 'identified',
        created_at: new Date(),
      })
      .returning();

    // 7. Registra predição no log
    await db.insert(prediction_log).values({
      lead_id: lead.id,
      product_id: pred.product_id,
      predicted_score: finalScore,
      predicted_days: pred.avg_days_to_purchase,
      created_at: new Date(),
    });

    // 8. Executa ação com base no score
    if (finalScore >= 75) {
      console.log(`[REVENUE] Score alto (${finalScore}%): Escalando SDR`);
      await executeAction(lead, pred, 'escalate_sdr', opportunity.id);
    } else if (finalScore >= 50) {
      console.log(`[REVENUE] Score médio (${finalScore}%): Enviando conteúdo`);
      await executeAction(lead, pred, 'send_content', opportunity.id);
    } else {
      console.log(`[REVENUE] Score baixo (${finalScore}%): Monitorando`);
      await executeAction(lead, pred, 'monitor', opportunity.id);
    }
  }
}

// ========== FUNÇÕES AUXILIARES ==========

function calculateIntentScore(signals: any[]): number {
  if (signals.length === 0) return 0;

  const weights: Record<string, number> = {
    google_search: 30,
    meta_click: 25,
    site_visit: 20,
    whatsapp_question: 40,
  };

  let total = 0;
  for (const signal of signals) {
    const weight = weights[signal.signal_type] || 10;
    total += weight * (signal.value / 100);
  }

  return Math.min(Math.round(total), 100);
}

function determineOpportunityType(lead: any, pred: any): 'upsell' | 'cross_sell' | 'renewal' | 'upgrade' {
  // Se o lead já comprou este produto antes, é renewal ou upgrade
  const alreadyBought = true; // Simplificado - buscar no purchase_history
  if (alreadyBought) {
    return pred.product_category === 'upgrade' ? 'upgrade' : 'renewal';
  }
  return 'cross_sell';
}

async function executeAction(lead: any, pred: any, action: string, opportunityId: number) {
  const product = await db.select().from(product_catalog).where(eq(product_catalog.id, pred.product_id)).limit(1);

  if (!product[0]) return;

  const productName = product[0].name;

  switch (action) {
    case 'escalate_sdr':
      await runSDR(lead.id, 'upsell_opportunity');
      await sendWhatsApp(
        lead.phone,
        `🚀 *Oportunidade de compra!*\n\nOlá ${lead.name}! Baseado na sua obra e nas suas pesquisas, identificamos que você pode precisar de *${productName}*.\n\nNossa equipe está pronta para ajudar com as melhores condições!`
      );
      await db.update(revenue_opportunities)
        .set({ action_taken: 'sdr_contacted', stage: 'contacted' })
        .where(eq(revenue_opportunities.id, opportunityId));
      break;

    case 'send_content':
      await sendWhatsApp(
        lead.phone,
        `📚 *Conteúdo Exclusivo!*\n\nOlá ${lead.name}! Separamos um material especial sobre *${productName}* para você.\n\nAcesse: [link do conteúdo]`
      );
      await db.update(revenue_opportunities)
        .set({ action_taken: 'content_sent', stage: 'contacted' })
        .where(eq(revenue_opportunities.id, opportunityId));
      break;

    case 'monitor':
    default:
      await db.update(revenue_opportunities)
        .set({ action_taken: 'monitor_only' })
        .where(eq(revenue_opportunities.id, opportunityId));
      break;
  }
}
