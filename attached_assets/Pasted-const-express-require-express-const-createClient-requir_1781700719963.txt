const express = require('express');
const { createClient } = require('@supabase/supabase-js');
const axios = require('axios');

const app = express();
app.use(express.json());

// 1. Conexão com o Supabase usando as Secrets do Replit
const supabaseUrl = process.env.SUPABASE_URL;
const supabaseKey = process.env.SUPABASE_SERVICE_ROLE_KEY; // Use a Service Role para operações de backend
const supabase = createClient(supabaseUrl, supabaseKey);

// Token de Verificação que você inventará e colocará no painel da Meta Developers
const WEBHOOK_VERIFY_TOKEN = process.env.META_VERIFY_TOKEN;

/**
 * =================================================================
 * ENDPOINT 1: VALIDAÇÃO DO WEBHOOK (Exigido pela Meta para ativar)
 * =================================================================
 */
app.get('/webhooks/whatsapp', (req, res) => {
  const mode = req.query['hub.mode'];
  const token = req.query['hub.verify_token'];
  const challenge = req.query['hub.challenge'];

  if (mode && token) {
    if (mode === 'subscribe' && token === WEBHOOK_VERIFY_TOKEN) {
      console.log('✅ Webhook verificado com sucesso pela Meta!');
      return res.status(200).send(challenge);
    }
    return res.status(403).sendStatus(403);
  }
  return res.sendStatus(400);
});

/**
 * =================================================================
 * ENDPOINT 2: RECEBIMENTO DE MENSAGENS E STATUS (WEBHOOK ATIVO)
 * =================================================================
 */
app.post('/webhooks/whatsapp', async (req, res) => {
  const body = req.body;

  // Garante que o payload veio do ecossistema do WhatsApp
  if (body.object === 'whatsapp_business_account') {
    try {
      const entry = body.entry?.[0];
      const waba_id = entry?.id; // ID da Conta Comercial da Empresa Cliente
      const changes = entry?.changes?.[0]?.value;

      if (!waba_id || !changes) return res.sendStatus(200);

      // Busca qual empresa do seu SaaS é dona deste waba_id no Supabase
      const { data: empresa, error } = await supabase
        .from('empresas')
        .select('id')
        .eq('waba_id', waba_id)
        .single();

      if (error || !empresa) {
        console.error(`❌ Evento recebido de um waba_id não cadastrado: ${waba_id}`);
        return res.sendStatus(200);
      }

      // Cenário A: O cliente respondeu ou enviou uma nova mensagem
      if (changes.messages) {
        const mensagem = changes.messages[0];
        const contato = changes.contacts?.[0];

        // Salva a mensagem recebida no histórico do chat da empresa correspondente
        await supabase.from('historico_chats').insert([{
          empresa_id: empresa.id,
          whatsapp_id: mensagem.id,
          cliente_fone: mensagem.from,
          cliente_nome: contato?.profile?.name || 'Cliente Anonimizado',
          tipo: mensagem.type,
          texto: mensagem.text?.body || null,
          direcao: 'entrada', // Entrada = Cliente enviou
          criado_em: new Date()
        }]);
        
        console.log(`📩 Nova mensagem guardada para a empresa ${empresa.id}`);
      }

      // Cenário B: Atualização de Status de entrega (enviado, entregue, lido)
      if (changes.statuses) {
        const status = changes.statuses[0];
        
        // Atualiza o status da mensagem enviada na sua tabela de logs
        await supabase
          .from('historico_chats')
          .update({ status_meta: status.status })
          .eq('whatsapp_id', status.id);
      }

    } catch (err) {
      console.error('❌ Erro ao processar payload do Webhook:', err.message);
    }
    
    // Meta exige resposta HTTP 200 rápida para não bloquear o fluxo
    return res.status(200).send('EVENT_RECEIVED');
  }

  return res.sendStatus(404);
});

/**
 * =================================================================
 * FUNÇÃO DE DISPARO ATIVO (Seu CRM chamará isso nas campanhas)
 * =================================================================
 */
async function enviarMensagemTemplate(empresaId, telefoneDestino, nomeTemplate, linguagem = 'pt_BR', componentes = []) {
  try {
    // 1. Busca as credenciais de criptografia/acesso exclusivas daquela empresa
    const { data: credenciais, error } = await supabase
      .from('empresas')
      .select('phone_number_id, access_token')
      .eq('id', empresaId)
      .single();

    if (error || !credenciais) throw new Error('Credenciais do WhatsApp não encontradas.');

    const { phone_number_id, access_token } = credenciais;

    // 2. Monta a requisição estruturada exigida pela Graph API v20.0+ da Meta
    const url = `https://facebook.com{phone_number_id}/messages`;
    
    const payload = {
      messaging_product: "whatsapp",
      to: telefoneDestino, // Ex: "5549999999999"
      type: "template",
      template: {
        name: nomeTemplate,
        language: {
          code: linguagen
        }
      }
    };

    // Adiciona variáveis personalizadas ao template se existirem (ex: Nome do lead, data)
    if (componentes.length > 0) {
      payload.template.components = componentes;
    }

    // 3. Faz o POST direto para a nuvem da Meta usando o token isolado da empresa
    const resposta = await axios.post(url, payload, {
      headers: {
        'Authorization': `Bearer ${access_token}`,
        'Content-Type': 'application/json'
      }
    });

    const metaData = resposta.data;

    // 4. Registra no banco do Supabase que a mensagem foi disparada com sucesso
    await supabase.from('historico_chats').insert([{
      empresa_id: empresaId,
      whatsapp_id: metaData.messages[0].id,
      cliente_fone: telefoneDestino,
      tipo: 'template',
      texto: `Template: ${nomeTemplate}`,
      direcao: 'saida', // Saída = Sua plataforma disparou
      status_meta: 'sent',
      criado_em: new Date()
    }]);

    return { sucesso: true, messageId: metaData.messages[0].id };

  } catch (err) {
    console.error(`❌ Falha no disparo da empresa ${empresaId}:`, err.response?.data || err.message);
    return { sucesso: false, erro: err.response?.data || err.message };
  }
}

/**
 * EXEMPLO DE ENDPOINT INTERNO DO SEU CRM PARA DISPARAR EM MASSA
 * Seu front-end enviará a lista de leads para cá
 */
app.post('/api/campanhas/disparar', async (req, res) => {
  const { empresa_id, leads, template_name } = req.body; 
  // leads = [{ fone: "5549...", nome: "João" }, ...]

  res.status(202).json({ mensagem: "Campanha iniciada. Processando na fila..." });

  // Executa o loop assincronamente com Delay para não travar o Replit (Fila Simples)
  for (const lead of leads) {
    // Exemplo de componente se seu template usar variáveis como: "Olá {{1}}"
    const componentes = [{
      type: "body",
      parameters: [{ type: "text", text: lead.nome }]
    }];

    await enviarMensagemTemplate(empresa_id, lead.fone, template_name, 'pt_BR', componentes);
    
    // Delay de 1,5 segundos (1500ms) entre mensagens para respeitar os limites iniciais da conta do cliente
    await new Promise(resolve => setTimeout(resolve, 1500));
  }
});

const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`🚀 SaaS rodando com sucesso na porta ${PORT}`);
});
