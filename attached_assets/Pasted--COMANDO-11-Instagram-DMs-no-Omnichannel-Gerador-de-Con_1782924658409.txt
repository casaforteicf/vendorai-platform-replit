### Parte B — Handler de mensagem Instagram

```javascript
async function handleInstagramMessage({ senderId, recipientId, message, timestamp }) {
  // 1. Buscar tenant pelo instagram_account_id
  const tenant = await db.query.tenants.findFirst({
    where: eq(tenants.instagram_account_id, recipientId)
  });
  if (!tenant) return;

  // 2. Buscar ou criar conversa
  let conversation = await db.query.conversations.findFirst({
    where: and(
      eq(conversations.tenantId, tenant.id),
      eq(conversations.channel, 'instagram'),
      eq(conversations.externalId, senderId)
    )
  });

  if (!conversation) {
    // Buscar perfil do Instagram do remetente via API
    const profile = await getInstagramProfile(senderId, tenant.instagram_access_token);
    
    conversation = await db.insert(conversations).values({
      tenantId: tenant.id,
      channel: 'instagram',
      externalId: senderId,
      contactName: profile?.name ?? 'Instagram User',
      contactUsername: profile?.username ?? null,
      contactAvatar: profile?.profile_pic ?? null,
      status: 'open'
    }).returning();
    conversation = conversation[0];
  }

  // 3. Salvar mensagem
  await db.insert(messages).values({
    conversationId: conversation.id,
    tenantId: tenant.id,
    direction: 'inbound',
    channel: 'instagram',
    content: message.text ?? '[mídia]',
    externalId: message.mid,
    sentAt: new Date(timestamp * 1000)
  });

  // 4. Notificar frontend via WebSocket/SSE
  await notifyNewMessage(tenant.id, conversation.id);
}
```

### Parte B — Backend: geração de conteúdo

```javascript
// POST /api/content/instagram/generate
async function gerarConteudoInstagram({ tenantId, tipo, tema, produtoId, tom }) {
  const tenant = await getTenantWithPlan(tenantId);
  const produto = produtoId ? await getProduto(produtoId) : null;

  const prompt = `
Você é um especialista em marketing para redes sociais de pequenas empresas brasileiras.

EMPRESA: ${tenant.name}
SEGMENTO: ${tenant.segment ?? 'varejo'}
TOM DE VOZ: ${tom}
TIPO DE CONTEÚDO: ${tipo} (${tipo === 'post' ? 'imagem estática' : tipo === 'story' ? 'story vertical' : 'vídeo curto'})
TEMA/OBJETIVO: ${tema}
${produto ? `PRODUTO EM DESTAQUE: ${produto.name} — R$ ${produto.price}` : ''}

Gere:

1. LEGENDA COMPLETA: texto para a legenda do ${tipo}, incluindo emojis relevantes, call-to-action claro e até 5 hashtags estratégicas para o segmento

2. DESCRIÇÃO VISUAL: descrição detalhada da imagem/vídeo ideal para esse conteúdo (o que mostrar, cores, enquadramento, elementos visuais)

3. SUGESTÃO DE HORÁRIO: melhor horário para postar esse tipo de conteúdo

4. TEXTO ALTERNATIVO: alt text acessível para a imagem

Seja específico para o segmento e região brasileira. Use linguagem natural, não corporativa.
`;

  const response = await openai.chat.completions.create({
    model: 'gpt-4o-mini',
    messages: [{ role: 'user', content: prompt }],
    max_tokens: 800
  });

  const conteudo = response.choices[0].message.content;

  // Salvar rascunho
  const rascunho = await db.insert(contentDrafts).values({
    tenantId,
    channel: 'instagram',
    tipo,
    tema,
    produtoId: produtoId ?? null,
    tom,
    conteudo,
    status: 'draft'
  }).returning();

  return { rascunho: rascunho[0], conteudo };
}
```

### Parte D — Agendamento de posts

```sql
CREATE TABLE IF NOT EXISTS content_drafts (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  tenant_id uuid REFERENCES tenants(id),
  channel text DEFAULT 'instagram',
  tipo text, -- post | story | reel
  tema text,
  produto_id uuid,
  tom text,
  conteudo text,
  legenda text,
  scheduled_for timestamptz,
  status text DEFAULT 'draft', -- draft | scheduled | published | cancelled
  published_at timestamptz,
  instagram_post_id text,
  created_at timestamptz DEFAULT now()
);
```
