🔍 DIAGNÓSTICO ATUAL (BASEADO NO SCHEDULER)

· Execução: churn-check no cron (a cada 5 minutos)
· Problemas:
  · ❌ Critérios fixos (ex: "sem interação há X dias") sem IA
  · ❌ Não usa dados de uso do sistema (frequência de login, ações realizadas)
  · ❌ Ações preventivas são genéricas (não personalizadas)
  · ❌ Não aprende com resultados (quem churnou mesmo após ação)
  · ❌ Sem score de risco dinâmico

🚀 MELHORIAS PROPOSTAS

1. Predição de Churn com IA — Analisa histórico de interações, frequência de uso, tickets de suporte, etc.
2. Score de Risco Dinâmico — Lead ganha score 0-100 (quanto maior, maior risco)
3. Ações Preventivas Automáticas — IA decide: oferecer desconto, enviar conteúdo de re-engajamento, escalar para vendedor sênior, ou apenas monitorar
4. Aprendizado com Resultados — Registra se o lead churnou e retroalimenta o modelo
5. Integração com SDR — Se risco for alto, dispara SDR para tentar reverter

CREATE TABLE IF NOT EXISTS churn_history (
  id SERIAL PRIMARY KEY,
  lead_id INTEGER NOT NULL REFERENCES leads(id),
  user_id INTEGER NOT NULL REFERENCES users(id), -- vendedor responsável
  churn_score FLOAT NOT NULL, -- 0-100
  risk_level TEXT CHECK (risk_level IN ('low', 'medium', 'high', 'critical')),
  predicted_at TIMESTAMP DEFAULT NOW(),
  churned_at TIMESTAMP, -- preenchido se o lead realmente cancelou
  features JSONB, -- dados usados para previsão
  action_taken TEXT, -- o que foi feito (ex: 'sent_discount', 'sent_content', 'escalated_sdr')
  action_successful BOOLEAN, -- se a ação reverteu o churn
  created_at TIMESTAMP DEFAULT NOW()
);

-- Tabela de ações preventivas (log)
CREATE TABLE IF NOT EXISTS churn_actions (
  id SERIAL PRIMARY KEY,
  lead_id INTEGER NOT NULL REFERENCES leads(id),
  action_type TEXT CHECK (action_type IN ('discount_offer', 'content_sent', 'escalated_sdr', 'call_scheduled', 'monitor_only')),
  message_sent TEXT,
  discount_amount FLOAT, -- se for desconto
  content_id INTEGER, -- se for conteúdo
  scheduled_at TIMESTAMP, -- se for agendado
  executed_at TIMESTAMP DEFAULT NOW(),
  status TEXT DEFAULT 'pending' CHECK (status IN ('pending', 'executed', 'failed', 'cancelled')),
  created_at TIMESTAMP DEFAULT NOW()
);

-- Adicionar colunas na tabela leads para churn
ALTER TABLE leads ADD COLUMN IF NOT EXISTS churn_risk_score FLOAT DEFAULT 0;
ALTER TABLE leads ADD COLUMN IF NOT EXISTS churn_risk_level TEXT CHECK (churn_risk_level IN ('low', 'medium', 'high', 'critical'));
ALTER TABLE leads ADD COLUMN IF NOT EXISTS last_activity_at TIMESTAMP;
ALTER TABLE leads ADD COLUMN IF NOT EXISTS churn_predicted_at TIMESTAMP;
ALTER TABLE leads ADD COLUMN IF NOT EXISTS churned_at TIMESTAMP;
ALTER TABLE leads ADD COLUMN IF NOT EXISTS is_churned BOOLEAN DEFAULT FALSE;

-- Índices para performance
CREATE INDEX IF NOT EXISTS idx_leads_churn_risk ON leads(churn_risk_score) WHERE is_churned = FALSE;
CREATE INDEX IF NOT EXISTS idx_leads_last_activity ON leads(last_activity_at) WHERE is_churned = FALSE;

// src/agents/churn-tools.ts
import { db } from '../db';
import { leads, churn_history, churn_actions, content_library } from '../schema';
import { eq, and, gte, lte, isNull, sql, desc, inArray } from 'drizzle-orm';
import { sendWhatsApp } from '../integrations/zapi';
import { runSDR } from './sdr-orchestrator';

// 1. Buscar leads em risco (com base em regras heurísticas + IA)
export const getLeadsAtRisk = async (tenantId: string) => {
  // Critérios heurísticos (rápidos):
  // - Sem atividade há mais de 7 dias
  // - Não abriu mensagens nas últimas 5 interações
  // - Estágio "Perdido" ou "Aguardando" há mais de 14 dias
  const now = new Date();
  const sevenDaysAgo = new Date(now.getTime() - 7 * 24 * 60 * 60 * 1000);
  const fourteenDaysAgo = new Date(now.getTime() - 14 * 24 * 60 * 60 * 1000);

  const leadsAtRisk = await db.select()
    .from(leads)
    .where(
      and(
        eq(leads.tenant_id, tenantId),
        eq(leads.is_churned, false),
        sql`
          (${leads.last_activity_at} IS NULL OR ${leads.last_activity_at} <= ${sevenDaysAgo.toISOString()}) OR
          (${leads.stage} IN ('Perdido', 'Aguardando') AND ${leads.updated_at} <= ${fourteenDaysAgo.toISOString()}) OR
          (${leads.churn_risk_score} >= 50)
        `
      )
    )
    .orderBy(desc(leads.churn_risk_score))
    .limit(50);

  return leadsAtRisk;
};

// 2. Calcular score de churn com base em dados do lead
export const calculateChurnScore = async (leadId: number): Promise<{ score: number; factors: string[] }> => {
  const lead = await db.select().from(leads).where(eq(leads.id, leadId)).limit(1);
  if (!lead[0]) return { score: 0, factors: [] };

  const factors: string[] = [];
  let score = 0;

  // Fator 1: Tempo sem atividade
  if (lead[0].last_activity_at) {
    const daysInactive = Math.floor((Date.now() - new Date(lead[0].last_activity_at).getTime()) / (1000 * 60 * 60 * 24));
    if (daysInactive > 30) { score += 30; factors.push('30+ dias inativo'); }
    else if (daysInactive > 14) { score += 20; factors.push('14+ dias inativo'); }
    else if (daysInactive > 7) { score += 10; factors.push('7+ dias inativo'); }
  } else {
    score += 20; factors.push('Sem atividade registrada');
  }

  // Fator 2: Estágio atual
  if (lead[0].stage === 'Perdido') { score += 30; factors.push('Estágio "Perdido"'); }
  else if (lead[0].stage === 'Aguardando') { score += 20; factors.push('Estágio "Aguardando"'); }
  else if (lead[0].stage === 'Interessado') { score += 5; factors.push('Estágio "Interessado"'); }

  // Fator 3: Qualidade do lead (baixa qualidade = maior risco)
  if (lead[0].quality_score && lead[0].quality_score < 30) { score += 20; factors.push('Qualidade baixa (<30)'); }
  else if (lead[0].quality_score && lead[0].quality_score < 50) { score += 10; factors.push('Qualidade média (30-50)'); }

  // Fator 4: No-shows (se já teve)
  if (lead[0].no_show_count && lead[0].no_show_count > 0) {
    score += Math.min(lead[0].no_show_count * 10, 30);
    factors.push(`${lead[0].no_show_count} no-show(s)`);
  }

  // Fator 5: Número de interações (poucas interações = maior risco)
  if (lead[0].total_appointments && lead[0].total_appointments === 0) {
    score += 10; factors.push('Nenhuma reunião agendada');
  }

  return { score: Math.min(score, 100), factors };
};

// 3. Registrar ação preventiva
export const logChurnAction = async (data: {
  leadId: number,
  actionType: 'discount_offer' | 'content_sent' | 'escalated_sdr' | 'call_scheduled' | 'monitor_only',
  messageSent?: string,
  discountAmount?: number,
  contentId?: number,
  scheduledAt?: Date,
}) => {
  const [action] = await db.insert(churn_actions)
    .values({
      lead_id: data.leadId,
      action_type: data.actionType,
      message_sent: data.messageSent,
      discount_amount: data.discountAmount,
      content_id: data.contentId,
      scheduled_at: data.scheduledAt,
      status: 'executed',
      executed_at: new Date(),
    })
    .returning();
  return action;
};

// 4. Enviar mensagem de retenção (personalizada)
export const sendRetentionMessage = async (leadId: number, message: string) => {
  const lead = await db.select().from(leads).where(eq(leads.id, leadId)).limit(1);
  if (!lead[0]) return;

  try {
    await sendWhatsApp(lead[0].phone, message);
    return { success: true };
  } catch (error) {
    console.error('[CHURN] Erro ao enviar mensagem de retenção:', error);
    return { success: false };
  }
};

// 5. Registrar churn (quando o lead realmente cancela)
export const registerChurn = async (leadId: number, userId: number) => {
  await db.update(leads)
    .set({ 
      is_churned: true,
      churned_at: new Date(),
      stage: 'Perdido'
    })
    .where(eq(leads.id, leadId));

  // Atualiza o histórico de churn (última predição)
  const [history] = await db.select()
    .from(churn_history)
    .where(
      and(
        eq(churn_history.lead_id, leadId),
        isNull(churn_history.churned_at)
      )
    )
    .orderBy(desc(churn_history.predicted_at))
    .limit(1);

  if (history) {
    await db.update(churn_history)
      .set({ churned_at: new Date() })
      .where(eq(churn_history.id, history.id));
  }

  console.log(`[CHURN] Lead ${leadId} registrado como churn.`);
};

// 6. Registrar resultado da ação (se foi bem-sucedida)
export const registerActionResult = async (actionId: number, successful: boolean) => {
  // Busca a ação
  const action = await db.select().from(churn_actions).where(eq(churn_actions.id, actionId)).limit(1);
  if (!action[0]) return;

  // Atualiza o histórico de churn com o resultado
  const [history] = await db.select()
    .from(churn_history)
    .where(
      and(
        eq(churn_history.lead_id, action[0].lead_id),
        isNull(churn_history.action_successful)
      )
    )
    .orderBy(desc(churn_history.predicted_at))
    .limit(1);

  if (history) {
    await db.update(churn_history)
      .set({ action_successful: successful })
      .where(eq(churn_history.id, history.id));
  }
};

// Schema para IA
export const churnToolSchemas = [

          score: { type: 'number', minimum: 0, maximum: 100, description: 'Score de risco (0-100)' },
          factors: { type: 'array', items: { type: 'string' }, description: 'Fatores que contribuem para o risco' },
          recommended_action: { type: 'string', enum: ['discount_offer', 'content_sent', 'escalated_sdr', 'call_scheduled', 'monitor_only'], description: 'Ação recomendada' },
          message_suggestion: { type: 'string', description: 'Sugestão de mensagem para enviar ao lead' },
          reasoning: { type: 'string', description: 'Explicação detalhada da análise' }
        },
        required: ['risk_level', 'score', 'factors', 'recommended_action', 'reasoning']
      }
    }
  }
];

// src/agents/churn-orchestrator.ts
import OpenAI from 'openai';
import { 
  getLeadsAtRisk,
  calculateChurnScore,
  logChurnAction,
  sendRetentionMessage,
  registerChurn,
  registerActionResult,
  churnToolSchemas
} from './churn-tools';
import { db } from '../db';
import { leads, churn_history, content_library } from '../schema';
import { eq } from 'drizzle-orm';
import { runSDR } from './sdr-orchestrator';

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

export const runChurnAgent = async (tenantId: string) => {
  console.log('[CHURN] Iniciando análise de churn...');

  // 1. Busca leads em risco
  const leadsAtRisk = await getLeadsAtRisk(tenantId);
  if (leadsAtRisk.length === 0) {
    console.log('[CHURN] Nenhum lead em risco.');
    return;
  }

  console.log(`[CHURN] ${leadsAtRisk.length} leads em risco.`);

  for (const lead of leadsAtRisk) {
    await processChurnRisk(lead, tenantId);
  }
};

const processChurnRisk = async (lead: any, tenantId: string) => {
  console.log(`[CHURN] Analisando lead ${lead.id} - ${lead.name}`);

  // 1. Calcula score heurístico
  const { score: heuristicScore, factors: heuristicFactors } = await calculateChurnScore(lead.id);

  // 2. Busca histórico de churn do lead (para contexto)
  const history = await db.select()
    .from(churn_history)
    .where(eq(churn_history.lead_id, lead.id))
    .orderBy('created_at', 'desc')
    .limit(5);

  // 3. Monta prompt para IA
  const systemPrompt = `
Você é um especialista em retenção de clientes B2B da VendorAI.
Analise este lead e determine o risco de churn.

Dados do Lead:
- Nome: ${lead.name}
- Empresa: ${lead.company || 'N/A'}
- Estágio: ${lead.stage}
- Qualidade: ${lead.quality_score || 'N/A'}
- Score heurístico: ${heuristicScore}/100
- Fatores heurísticos: ${heuristicFactors.join(', ')}
- Total de reuniões: ${lead.total_appointments || 0}
- No-shows: ${lead.no_show_count || 0}

Histórico de predições anteriores: ${history.length > 0 ? history.map(h => `Score ${h.churn_score} em ${h.predicted_at}`).join('\n') : 'Nenhum'}

Regras:
- Se score > 80: risco CRITICAL → ação: escalar para SDR (escalated_sdr) imediatamente
- Se score > 60: risco HIGH → ação: oferecer desconto (discount_offer) ou agendar call
- Se score > 40: risco MEDIUM → ação: enviar conteúdo de re-engajamento (content_sent)
- Se score < 40: risco LOW → ação: apenas monitorar (monitor_only)

Use a ferramenta "analyzeChurnRisk" para dar sua análise e recomendação.
`;

  const response = await openai.chat.completions.create({
    model: 'gpt-4o-mini',
    messages: [
      { role: 'system', content: systemPrompt },
      { role: 'user', content: `Analise o lead ${lead.name} e recomende ação preventiva.` }
    ],
    tools: churnToolSchemas,
    tool_choice: { type: 'function', function: { name: 'analyzeChurnRisk' } },
    temperature: 0.3,
  });

  const toolCall = response.choices[0].message.tool_calls?.[0];
  if (!toolCall) {
    console.log(`[CHURN] IA não retornou análise para lead ${lead.id}`);
    return;
  }

  const args = JSON.parse(toolCall.function.arguments);
  console.log(`[CHURN] Análise: Risk ${args.risk_level}, Score ${args.score}/100`);

  // 4. Salva no histórico
  await db.insert(churn_history).values({
    lead_id: lead.id,
    user_id: lead.user_id || 1,
    churn_score: args.score,
    risk_level: args.risk_level,
    features: {
      heuristicScore,
      heuristicFactors,
      stage: lead.stage,
      qualityScore: lead.quality_score,
      totalAppointments: lead.total_appointments,
      noShowCount: lead.no_show_count,
    },
    action_taken: args.recommended_action,
    predicted_at: new Date(),
  });

  // 5. Atualiza lead com score
  await db.update(leads)
    .set({
      churn_risk_score: args.score,
      churn_risk_level: args.risk_level,
      churn_predicted_at: new Date(),
    })
    .where(eq(leads.id, lead.id));

  // 6. Executa ação recomendada
  await executeChurnAction(lead, args);
};

const executeChurnAction = async (lead: any, analysis: any) => {
  const actionType = analysis.recommended_action;
  const message = analysis.message_suggestion || '';

  console.log(`[CHURN] Executando ação: ${actionType} para lead ${lead.id}`);

  switch (actionType) {
    case 'discount_offer':
      // Oferece desconto
      const discountMsg = `🎯 *Oferta Especial para Você!*\n\nOlá ${lead.name}! Estamos com uma promoção exclusiva para clientes que estão conosco desde o início.\n\n💰 *Desconto de 20%* no próximo mês!\n\nAproveite essa oportunidade para continuar transformando suas vendas com a VendorAI.`;
      await sendRetentionMessage(lead.id, discountMsg);
      await logChurnAction({
        leadId: lead.id,
        actionType: 'discount_offer',
        messageSent: discountMsg,
        discountAmount: 20,
      });
      break;

    case 'content_sent':
      // Busca conteúdo de re-engajamento
      const content = await db.select()
        .from(content_library)
        .where(sql`${content_library.tags} && ARRAY['reengajamento', 'retencao']`)
        .orderBy('engagement_score', 'desc')
        .limit(1);

      if (content[0]) {
        const contentMsg = `📚 *Conteúdo Exclusivo para Você!*\n\nOlá ${lead.name}! Separamos um conteúdo especial que pode ajudar você a maximizar seus resultados com a VendorAI.\n\n📌 ${content[0].title}\n${content[0].description || ''}\n\n🔗 Acesse: ${content[0].url}`;
        await sendRetentionMessage(lead.id, contentMsg);
        await logChurnAction({
          leadId: lead.id,
          actionType: 'content_sent',
          messageSent: contentMsg,
          contentId: content[0].id,
        });
      } else {
        // Fallback: mensagem genérica
        const fallbackMsg = `👋 Olá ${lead.name}! Faz um tempo que não falamos. Queremos saber como podemos ajudar você a ter ainda mais sucesso com a VendorAI. Estamos aqui para você!`;
        await sendRetentionMessage(lead.id, fallbackMsg);
        await logChurnAction({
          leadId: lead.id,
          actionType: 'content_sent',
          messageSent: fallbackMsg,
        });
      }
      break;

    case 'escalated_sdr':
      // Escala para SDR (dispara o SDR para tentar reverter)
      const sdrMessage = `🔴 *Lead em Risco de Churn!*\n\nOlá! O lead ${lead.name} da empresa ${lead.company || 'N/A'} está com alto risco de churn (${analysis.score}%).\n\nPor favor, entre em contato imediatamente para tentar reverter.`;
      
      // Dispara SDR para este lead (com evento especial)
      await runSDR(lead.id, 'churn_alert');
      
      // Envia mensagem para o lead (ação imediata)
      const escalationMsg = `⚠️ *Estamos aqui para ajudar!*\n\nOlá ${lead.name}! Percebemos que você pode estar enfrentando algum desafio com a nossa plataforma.\n\nNosso time de especialistas está pronto para conversar com você e encontrar a melhor solução. Você pode agendar uma reunião conosco agora mesmo.`;
      await sendRetentionMessage(lead.id, escalationMsg);
      
      await logChurnAction({
        leadId: lead.id,
        actionType: 'escalated_sdr',
        messageSent: escalationMsg,
      });
      break;

    case 'call_scheduled':
      // Agenda call de retenção (integra com Scheduler)
      const callMsg = `📞 *Vamos conversar!*\n\nOlá ${lead.name}! Agende uma call com nosso time de sucesso do cliente para entendermos melhor suas necessidades.\n\nClique no link para escolher o melhor horário: [link de agendamento]`;
      await sendRetentionMessage(lead.id, callMsg);
      await logChurnAction({
        leadId: lead.id,
        actionType: 'call_scheduled',
        messageSent: callMsg,
        scheduledAt: new Date(Date.now() + 2 * 24 * 60 * 60 * 1000), // 2 dias
      });
      break;

    case 'monitor_only':
    default:
      // Apenas monitora (não faz ação)
      await logChurnAction({
        leadId: lead.id,
        actionType: 'monitor_only',
        messageSent: 'Apenas monitorando',
      });
      console.log(`[CHURN] Lead ${lead.id} em monitoramento (risco ${analysis.risk_level}).`);
      break;
  }
};

// src/routes/churn-webhook.ts
import express from 'express';
import { registerChurn, registerActionResult } from '../agents/churn-tools';

const router = express.Router();

// Endpoint para registrar que um lead churnou (cancelou)
router.post('/api/churn/register', async (req, res) => {
  try {
    const { leadId, userId } = req.body;
    if (!leadId) {
      return res.status(400).json({ error: 'leadId é obrigatório' });
    }
    await registerChurn(parseInt(leadId), parseInt(userId || '1'));
    res.json({ success: true });
  } catch (error) {
    console.error('[CHURN WEBHOOK] Erro:', error);
    res.status(500).json({ error: 'Internal error' });
  }
});

// Endpoint para registrar se uma ação foi bem-sucedida (reverteu o churn)
router.post('/api/churn/action-result', async (req, res) => {
  try {
    const { actionId, successful } = req.body;
    if (!actionId) {
      return res.status(400).json({ error: 'actionId é obrigatório' });
    }
    await registerActionResult(parseInt(actionId), successful);
    res.json({ success: true });
  } catch (error) {
    console.error('[CHURN WEBHOOK] Erro:', error);
    res.status(500).json({ error: 'Internal error' });
  }
});

export default router;
