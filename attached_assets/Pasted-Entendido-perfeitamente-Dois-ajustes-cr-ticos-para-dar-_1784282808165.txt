```sql
-- ========================================
-- 1. TABELA DE MÓDULOS DISPONÍVEIS (CATÁLOGO GLOBAL)
-- ========================================
CREATE TABLE IF NOT EXISTS available_modules (
  id SERIAL PRIMARY KEY,
  module_key TEXT UNIQUE NOT NULL, -- ex: 'dashboard', 'leads', 'sdr', 'prospector', 'content', 'campaigns', 'scheduler', 'revenue', 'churn', 'settings'
  module_name TEXT NOT NULL, -- ex: 'Dashboard', 'Leads', 'SDR', 'Prospector', 'Conteúdo', 'Campanhas', 'Agendamentos', 'Receita', 'Churn', 'Configurações'
  icon TEXT, -- Nome do ícone (ex: 'LayoutDashboard', 'Users', 'Bot', 'Search', 'FileText', 'Megaphone', 'Calendar', 'DollarSign', 'AlertCircle', 'Settings')
  description TEXT,
  is_default BOOLEAN DEFAULT FALSE, -- Se é ativado por padrão para novos tenants
  created_at TIMESTAMP DEFAULT NOW()
);

-- ========================================
-- 2. TABELA DE MÓDULOS POR TENANT
-- ========================================
CREATE TABLE IF NOT EXISTS tenant_modules (
  id SERIAL PRIMARY KEY,
  tenant_id UUID NOT NULL REFERENCES tenants(id) ON DELETE CASCADE,
  module_key TEXT NOT NULL REFERENCES available_modules(module_key) ON DELETE CASCADE,
  is_enabled BOOLEAN DEFAULT TRUE,
  updated_at TIMESTAMP DEFAULT NOW(),
  UNIQUE(tenant_id, module_key)
);

CREATE INDEX idx_tenant_modules_tenant ON tenant_modules(tenant_id);

-- ========================================
-- 3. ADICIONAR CAMPO DE COMISSÃO NO CATÁLOGO DE PRODUTOS
-- ========================================
ALTER TABLE product_catalog ADD COLUMN IF NOT EXISTS commission_percentage DECIMAL(5,2); -- Ex: 5.00 = 5%

-- ========================================
-- 4. ADICIONAR CAMPO DE COMISSÃO PADRÃO NO TENANT (OPCIONAL)
-- ========================================
ALTER TABLE tenants ADD COLUMN IF NOT EXISTS default_commission_percentage DECIMAL(5,2) DEFAULT 0;

-- ========================================
-- 5. SEED DOS MÓDULOS DISPONÍVEIS (INSERIR OS PADRÕES)
-- ========================================
INSERT INTO available_modules (module_key, module_name, icon, description, is_default) VALUES
  ('dashboard', 'Dashboard', 'LayoutDashboard', 'Visão geral do negócio', TRUE),
  ('leads', 'Leads', 'Users', 'Gestão de leads e contatos', TRUE),
  ('sdr', 'SDR', 'Bot', 'Atendimento e qualificação', TRUE),
  ('prospector', 'Prospector', 'Search', 'Busca ativa de leads', FALSE),
  ('content', 'Conteúdo', 'FileText', 'Nutrição e fluxos de conteúdo', FALSE),
  ('campaigns', 'Campanhas', 'Megaphone', 'Disparo em lote', FALSE),
  ('scheduler', 'Agendamentos', 'Calendar', 'Reuniões e lembretes', TRUE),
  ('revenue', 'Receita', 'DollarSign', 'Oportunidades preditivas', FALSE),
  ('churn', 'Churn', 'AlertCircle', 'Prevenção de cancelamento', FALSE),
  ('settings', 'Configurações', 'Settings', 'Configurações do sistema', TRUE)
ON CONFLICT (module_key) DO NOTHING;

-- ========================================
-- 6. ATIVAR MÓDULOS PADRÃO PARA TENANTS EXISTENTES
-- ========================================
INSERT INTO tenant_modules (tenant_id, module_key, is_enabled)
SELECT 
  t.id,
  m.module_key,
  m.is_default
FROM tenants t
CROSS JOIN available_modules m
WHERE m.is_default = TRUE
ON CONFLICT (tenant_id, module_key) DO NOTHING;
```

```typescript
// src/routes/tenant-modules.ts
import express from 'express';
import { db } from '../db';
import { available_modules, tenant_modules } from '../schema';
import { eq, and } from 'drizzle-orm';
import { authenticate } from '../middleware/auth';
import { authorize } from '../middleware/authorize';

const router = express.Router();

// 1. LISTAR TODOS OS MÓDULOS DISPONÍVEIS (Catálogo Global)
router.get('/api/modules/available', authenticate, async (req, res) => {
  try {
    const modules = await db.select().from(available_modules).orderBy(available_modules.module_name);
    res.json(modules);
  } catch (error) {
    console.error('[MODULES] Erro:', error);
    res.status(500).json({ error: 'Internal error' });
  }
});

// 2. LISTAR MÓDULOS ATIVOS DO TENANT
router.get('/api/tenants/:tenantId/modules', authenticate, async (req, res) => {
  try {
    const { tenantId } = req.params;
    
    const modules = await db.select()
      .from(tenant_modules)
      .innerJoin(available_modules, eq(available_modules.module_key, tenant_modules.module_key))
      .where(eq(tenant_modules.tenant_id, tenantId))
      .orderBy(available_modules.module_name);
    
    res.json(modules);
  } catch (error) {
    console.error('[MODULES] Erro:', error);
    res.status(500).json({ error: 'Internal error' });
  }
});

// 3. ATUALIZAR MÓDULOS DO TENANT (ATIVAR/DESATIVAR) - Apenas Owner/Admin
router.put('/api/tenants/:tenantId/modules', authenticate, authorize('Owner', 'Admin'), async (req, res) => {
  try {
    const { tenantId } = req.params;
    const { modules } = req.body; // Array de { module_key, is_enabled }

    if (!modules || !Array.isArray(modules)) {
      return res.status(400).json({ error: 'modules deve ser um array' });
    }

    // Para cada módulo, faz upsert
    for (const mod of modules) {
      const { module_key, is_enabled } = mod;
      
      await db.insert(tenant_modules)
        .values({
          tenant_id: tenantId,
          module_key: module_key,
          is_enabled: is_enabled,
          updated_at: new Date(),
        })
        .onConflictDoUpdate({
          target: [tenant_modules.tenant_id, tenant_modules.module_key],
          set: { is_enabled: is_enabled, updated_at: new Date() },
        });
    }

    res.json({ success: true });
  } catch (error) {
    console.error('[MODULES] Erro:', error);
    res.status(500).json({ error: 'Internal error' });
  }
});

// 4. ATIVAR/DESATIVAR UM MÓDULO ESPECÍFICO
router.patch('/api/tenants/:tenantId/modules/:moduleKey', authenticate, authorize('Owner', 'Admin'), async (req, res) => {
  try {
    const { tenantId, moduleKey } = req.params;
    const { is_enabled } = req.body;

    if (typeof is_enabled !== 'boolean') {
      return res.status(400).json({ error: 'is_enabled deve ser booleano' });
    }

    await db.insert(tenant_modules)
      .values({
        tenant_id: tenantId,
        module_key: moduleKey,
        is_enabled: is_enabled,
        updated_at: new Date(),
      })
      .onConflictDoUpdate({
        target: [tenant_modules.tenant_id, tenant_modules.module_key],
        set: { is_enabled: is_enabled, updated_at: new Date() },
      });

    res.json({ success: true });
  } catch (error) {
    console.error('[MODULES] Erro:', error);
    res.status(500).json({ error: 'Internal error' });
  }
});

export default router;
```

```typescript
// src/routes/product-commission.ts
import express from 'express';
import { db } from '../db';
import { product_catalog } from '../schema';
import { eq } from 'drizzle-orm';
import { authenticate } from '../middleware/auth';
import { authorize } from '../middleware/authorize';

const router = express.Router();

// 1. ATUALIZAR COMISSÃO DE UM PRODUTO
router.patch('/api/products/:productId/commission', authenticate, authorize('Owner', 'Admin'), async (req, res) => {
  try {
    const { productId } = req.params;
    const { commission_percentage } = req.body;

    if (commission_percentage === undefined || commission_percentage === null) {
      return res.status(400).json({ error: 'commission_percentage é obrigatório' });
    }

    if (commission_percentage < 0 || commission_percentage > 100) {
      return res.status(400).json({ error: 'commission_percentage deve estar entre 0 e 100' });
    }

    await db.update(product_catalog)
      .set({ commission_percentage: commission_percentage })
      .where(eq(product_catalog.id, parseInt(productId)));

    res.json({ success: true });
  } catch (error) {
    console.error('[COMMISSION] Erro:', error);
    res.status(500).json({ error: 'Internal error' });
  }
});

// 2. BUSCAR COMISSÃO DE UM PRODUTO
router.get('/api/products/:productId/commission', authenticate, async (req, res) => {
  try {
    const { productId } = req.params;
    const product = await db.select()
      .from(product_catalog)
      .where(eq(product_catalog.id, parseInt(productId)))
      .limit(1);

    if (!product[0]) {
      return res.status(404).json({ error: 'Produto não encontrado' });
    }

    res.json({ commission_percentage: product[0].commission_percentage || null });
  } catch (error) {
    console.error('[COMMISSION] Erro:', error);
    res.status(500).json({ error: 'Internal error' });
  }
});

// 3. ATUALIZAR COMISSÃO PADRÃO DO TENANT
router.patch('/api/tenants/:tenantId/default-commission', authenticate, authorize('Owner', 'Admin'), async (req, res) => {
  try {
    const { tenantId } = req.params;
    const { default_commission_percentage } = req.body;

    if (default_commission_percentage === undefined || default_commission_percentage === null) {
      return res.status(400).json({ error: 'default_commission_percentage é obrigatório' });
    }

    if (default_commission_percentage < 0 || default_commission_percentage > 100) {
      return res.status(400).json({ error: 'default_commission_percentage deve estar entre 0 e 100' });
    }

    await db.update(tenants)
      .set({ default_commission_percentage: default_commission_percentage })
      .where(eq(tenants.id, tenantId));

    res.json({ success: true });
  } catch (error) {
    console.error('[COMMISSION] Erro:', error);
    res.status(500).json({ error: 'Internal error' });
  }
});

export default router;
```

```typescript
// src/services/commission-calculator.ts
import { db } from '../db';
import { product_catalog, tenants, users } from '../schema';
import { eq } from 'drizzle-orm';

interface CommissionParams {
  productId: number;
  sellerId: number;
  tenantId: string;
  saleValue: number;
}

export const calculateCommission = async (params: CommissionParams): Promise<{ percentage: number; amount: number }> => {
  const { productId, sellerId, tenantId, saleValue } = params;

  // 1. Busca o produto e sua comissão específica
  const product = await db.select()
    .from(product_catalog)
    .where(eq(product_catalog.id, productId))
    .limit(1);

  let commissionPercentage = 0;

  if (product[0] && product[0].commission_percentage !== null && product[0].commission_percentage !== undefined) {
    // PRIORIDADE 1: Comissão específica do produto
    commissionPercentage = product[0].commission_percentage;
    console.log(`[COMMISSION] Usando comissão do produto: ${commissionPercentage}%`);
  } else {
    // PRIORIDADE 2: Comissão do vendedor (se existir)
    const seller = await db.select()
      .from(users)
      .where(eq(users.id, sellerId))
      .limit(1);

    if (seller[0] && seller[0].commission_percentage !== null && seller[0].commission_percentage !== undefined) {
      commissionPercentage = seller[0].commission_percentage;
      console.log(`[COMMISSION] Usando comissão do vendedor: ${commissionPercentage}%`);
    } else {
      // PRIORIDADE 3: Comissão padrão do tenant
      const tenant = await db.select()
        .from(tenants)
        .where(eq(tenants.id, tenantId))
        .limit(1);

      if (tenant[0] && tenant[0].default_commission_percentage !== null && tenant[0].default_commission_percentage !== undefined) {
        commissionPercentage = tenant[0].default_commission_percentage;
        console.log(`[COMMISSION] Usando comissão padrão do tenant: ${commissionPercentage}%`);
      } else {
        // PRIORIDADE 4: Padrão global (0% ou valor fixo)
        commissionPercentage = 0;
        console.log(`[COMMISSION] Nenhuma comissão configurada.`);
      }
    }
  }

  const amount = (saleValue * commissionPercentage) / 100;
  return { percentage: commissionPercentage, amount: Math.round(amount * 100) / 100 };
};
```

```typescript
// src/hooks/useTenantModules.ts
import { useEffect, useState } from 'react';
import axios from 'axios';

interface Module {
  module_key: string;
  module_name: string;
  icon: string;
  description: string;
  is_enabled: boolean;
}

export const useTenantModules = (tenantId: string) => {
  const [modules, setModules] = useState<Module[]>([]);
  const [loading, setLoading] = useState(true);
  const [error, setError] = useState<string | null>(null);

  useEffect(() => {
    const fetchModules = async () => {
      try {
        const response = await axios.get(`/api/tenants/${tenantId}/modules`);
        setModules(response.data);
      } catch (err: any) {
        setError(err.message);
      } finally {
        setLoading(false);
      }
    };

    if (tenantId) fetchModules();
  }, [tenantId]);

  // Função para verificar se um módulo está ativo
  const isModuleEnabled = (moduleKey: string): boolean => {
    const mod = modules.find(m => m.module_key === moduleKey);
    return mod ? mod.is_enabled : false;
  };

  // Função para atualizar um módulo
  const toggleModule = async (moduleKey: string, isEnabled: boolean) => {
    try {
      await axios.patch(`/api/tenants/${tenantId}/modules/${moduleKey}`, { is_enabled: isEnabled });
      setModules(prev =>
        prev.map(m =>
          m.module_key === moduleKey ? { ...m, is_enabled: isEnabled } : m
        )
      );
    } catch (err: any) {
      console.error('[MODULES] Erro ao atualizar:', err);
      throw err;
    }
  };

  return { modules, loading, error, isModuleEnabled, toggleModule };
};
```
