· Execução: campaigns no cron (a cada 15 minutos) — "envia campanhas vencidas"
· Problemas:
  · ❌ Disparo em massa baseado apenas em data de vencimento (sem segmentação)
  · ❌ Sem otimização de horário (todos recebem no mesmo momento)
  · ❌ Sem A/B testing de mensagens
  · ❌ Sem análise de performance pós-disparo (abertura, clique, conversão)
  · ❌ Sem inteligência para escolher a melhor abordagem para cada lead

🚀 MELHORIAS PROPOSTAS

1. Segmentação Inteligente — IA filtra leads com maior propensão a converter para cada campanha
2. Otimização de Horário — Envia no melhor horário com base no histórico de interação do lead
3. A/B Testing — IA gera 2 variações de mensagem, testa em amostras, escolhe a vencedora
4. Análise de Performance — Registra aberturas, cliques e conversões para ajustar campanhas futuras

```typescript
// src/agents/campaign-tools.ts
import { db } from '../db';
import { campaigns, leads, campaign_sent_logs, lead_optimal_hours } from '../schema';
import { eq, and, gte, lte, isNull, sql, inArray, desc, count } from 'drizzle-orm';
import { sendWhatsApp } from '../integrations/zapi';

// 1. Buscar campanhas ativas prontas para disparo
export const getActiveCampaigns = async (tenantId: string) => {
  const now = new Date();
  const campaignsList = await db.select()
    .from(campaigns)
    .where(
      and(
        eq(campaigns.tenant_id, tenantId),
        eq(campaigns.is_active, true),
        sql`${campaigns.scheduled_at} <= ${now.toISOString()}`,
        sql`${campaigns.sent_count} < ${campaigns.max_sends} OR ${campaigns.max_sends} IS NULL`
      )
    )
    .orderBy(desc(campaigns.priority));
  
  return campaignsList;
};

// 2. Buscar leads segmentados para uma campanha (usando critérios da IA)
export const getSegmentedLeads = async (campaignId: number, criteria: any, limit: number = 100) => {
  let query = db.select()
    .from(leads)
    .where(
      and(
        eq(leads.tenant_id, criteria.tenant_id),
        eq(leads.campaign_opt_out, false),
        isNull(leads.deleted_at),
        // Não enviar para quem já recebeu campanha nas últimas 48h
        sql`${leads.last_campaign_received_at} IS NULL OR ${leads.last_campaign_received_at} < NOW() - INTERVAL '48 hours'`
      )
    );

  // Aplica filtros do critério
  if (criteria.stage) {
    query = query.where(eq(leads.stage, criteria.stage));
  }
  if (criteria.industry) {
    query = query.where(eq(leads.industry, criteria.industry));
  }
  if (criteria.min_quality_score) {
    query = query.where(sql`${leads.quality_score} >= ${criteria.min_quality_score}`);
  }
  if (criteria.has_phone) {
    query = query.where(sql`${leads.phone} IS NOT NULL AND ${leads.phone} != ''`);
  }
  // Ordena por leads com maior propensão (score de qualidade)
  query = query.orderBy(desc(leads.quality_score)).limit(limit);

  return query;
};

// 3. Registrar envio de campanha
export const logCampaignSent = async (campaignId: number, leadId: number, variant: string, message: string) => {
  const [log] = await db.insert(campaign_sent_logs)
    .values({
      campaign_id: campaignId,
      lead_id: leadId,
      variant: variant as 'A' | 'B',
      message_text: message,
      sent_at: new Date()
    })
    .returning();

  // Atualiza contagem de envios da campanha
  await db.update(campaigns)
    .set({ sent_count: sql`${campaigns.sent_count} + 1` })
    .where(eq(campaigns.id, campaignId));

  // Atualiza último recebimento do lead
  await db.update(leads)
    .set({ last_campaign_received_at: new Date() })
    .where(eq(leads.id, leadId));

  return log;
};

// 4. Registrar interação (abertura, clique, conversão)
export const logCampaignInteraction = async (logId: number, type: 'opened' | 'clicked' | 'converted') => {
  const updateData: any = {};
  const campaignUpdate: any = {};

  if (type === 'opened') {
    updateData.opened_at = new Date();
    campaignUpdate.opened_count = sql`${campaigns.opened_count} + 1`;
  } else if (type === 'clicked') {
    updateData.clicked_at = new Date();
    campaignUpdate.clicked_count = sql`${campaigns.clicked_count} + 1`;
  } else if (type === 'converted') {
    updateData.converted_at = new Date();
    campaignUpdate.converted_count = sql`${campaigns.converted_count} + 1`;
  }

  await db.update(campaign_sent_logs)
    .set(updateData)
    .where(eq(campaign_sent_logs.id, logId));

  // Atualiza estatísticas da campanha
  const log = await db.select().from(campaign_sent_logs).where(eq(campaign_sent_logs.id, logId)).limit(1);
  if (log[0]) {
    await db.update(campaigns)
      .set(campaignUpdate)
      .where(eq(campaigns.id, log[0].campaign_id));
  }
};

// 5. Buscar horário ótimo para um lead
export const getOptimalHour = async (leadId: number): Promise<number | null> => {
  const [record] = await db.select()
    .from(lead_optimal_hours)
    .where(eq(lead_optimal_hours.lead_id, leadId))
    .limit(1);
  
  return record?.optimal_hour ?? null;
};

// 6. Atualizar horário ótimo com base no histórico
export const updateOptimalHour = async (leadId: number, interactions: any[]) => {
  // Análise simples: hora com mais interações (aberturas)
  if (interactions.length < 5) return;

  const hourCount: Record<number, number> = {};
  for (const interaction of interactions) {
    if (interaction.opened_at) {
      const hour = new Date(interaction.opened_at).getHours();
      hourCount[hour] = (hourCount[hour] || 0) + 1;
    }
  }

  let bestHour = 9; // padrão 9h
  let bestCount = 0;
  for (const [hour, count] of Object.entries(hourCount)) {
    if (count > bestCount) {
      bestCount = count;
      bestHour = parseInt(hour);
    }
  }

  const confidence = interactions.length > 20 ? 0.9 : 0.5;
  
  await db.insert(lead_optimal_hours)
    .values({ lead_id: leadId, optimal_hour: bestHour, confidence: confidence })
    .onConflictDoUpdate({
      target: lead_optimal_hours.lead_id,
      set: { optimal_hour: bestHour, confidence: confidence, last_updated: new Date() }
    });
};

// Schemas para IA
export const campaignToolSchemas = [

      parameters: {
        type: 'object',
        properties: {
          variant_a: { type: 'string', description: 'Mensagem da variante A (ex: abordagem mais direta)' },
          variant_b: { type: 'string', description: 'Mensagem da variante B (ex: abordagem mais consultiva)' },
          recommended_variant: { type: 'string', enum: ['A', 'B'], description: 'Qual variante é recomendada inicialmente' },
          reasoning: { type: 'string', description: 'Motivo da recomendação' }
        },
        required: ['variant_a', 'variant_b', 'recommended_variant', 'reasoning']
      }
    }
  },
  {
    type: 'function',
    function: {
      name: 'selectLeadsForCampaign',
      description: 'Define os critérios de segmentação para esta campanha.',
      parameters: {
        type: 'object',
        properties: {
          stage: { type: 'string', enum: ['Novo', 'Interessado', 'Qualificado', 'Aguardando', 'Perdido'] },
          industry: { type: 'string', description: 'Setor alvo (ex: Imobiliario, Fintech)' },
          min_quality_score: { type: 'number', minimum: 0, maximum: 100, description: 'Score mínimo de qualidade' },
```

```typescript
// src/agents/campaign-orchestrator.ts
import OpenAI from 'openai';
import { 
  getActiveCampaigns,
  getSegmentedLeads,
  logCampaignSent,
  logCampaignInteraction,
  getOptimalHour,
  updateOptimalHour,
  campaignToolSchemas
} from './campaign-tools';
import { db } from '../db';
import { campaigns, leads, campaign_sent_logs, content_interactions } from '../schema';
import { eq, and, desc, inArray, sql } from 'drizzle-orm';
import { sendWhatsApp } from '../integrations/zapi';

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });

export const runCampaignAgent = async (tenantId: string) => {
  console.log('[CAMPAIGN] Iniciando ciclo de campanhas...');

  // 1. Busca campanhas ativas
  const activeCampaigns = await getActiveCampaigns(tenantId);
  if (activeCampaigns.length === 0) {
    console.log('[CAMPAIGN] Nenhuma campanha ativa.');
    return;
  }

  console.log(`[CAMPAIGN] ${activeCampaigns.length} campanhas ativas.`);

  for (const campaign of activeCampaigns) {
    await processCampaign(campaign);
  }
};

const processCampaign = async (campaign: any) => {
  console.log(`[CAMPAIGN] Processando campanha: ${campaign.name}`);

  // 1. Define critérios de segmentação (IA ou configuração estática)
  let segmentCriteria = campaign.segment_criteria || { stage: 'Interessado' };
  
  // Se não tiver critérios, usa IA para sugerir
  if (!campaign.segment_criteria || Object.keys(campaign.segment_criteria).length === 0) {
    const criteriaResponse = await openai.chat.completions.create({
      model: 'gpt-4o-mini',
      messages: [
        { 
          role: 'system', 
          content: `Defina critérios de segmentação para a campanha "${campaign.name}". 
          Descrição: ${campaign.description || 'Campanha de vendas'}.
          Sugira estágio, setor e score mínimo. Retorne apenas o JSON.`
        }
      ],
      tools: campaignToolSchemas.filter(t => t.function.name === 'selectLeadsForCampaign'),
      tool_choice: { type: 'function', function: { name: 'selectLeadsForCampaign' } },
      temperature: 0.3,
    });

    const toolCall = criteriaResponse.choices[0].message.tool_calls?.[0];
    if (toolCall) {
      segmentCriteria = JSON.parse(toolCall.function.arguments);
      // Salva os critérios na campanha para uso futuro
      await db.update(campaigns)
        .set({ segment_criteria: segmentCriteria })
        .where(eq(campaigns.id, campaign.id));
    }
  }

  // 2. Busca leads segmentados
  const leadsList = await getSegmentedLeads(
    campaign.id, 
    { ...segmentCriteria, tenant_id: campaign.tenant_id },
    campaign.max_sends || 100
  );

  if (leadsList.length === 0) {
    console.log(`[CAMPAIGN] Nenhum lead encontrado para campanha ${campaign.id}`);
    return;
  }

  console.log(`[CAMPAIGN] ${leadsList.length} leads qualificados para campanha ${campaign.id}`);

  // 3. Gera mensagens personalizadas (IA) para cada lead
  for (const lead of leadsList) {
    await sendCampaignMessage(campaign, lead);
  }

  console.log(`[CAMPAIGN] Campanha ${campaign.id} processada.`);
};

const sendCampaignMessage = async (campaign: any, lead: any) => {
  console.log(`[CAMPAIGN] Enviando para lead ${lead.id} - ${lead.name}`);

  // 1. Busca horário ótimo
  const optimalHour = await getOptimalHour(lead.id);
  const currentHour = new Date().getHours();
  
  // Se o lead tem horário ótimo e não é o horário, agenda para depois
  if (optimalHour !== null && Math.abs(currentHour - optimalHour) > 2) {
    console.log(`[CAMPAIGN] Lead ${lead.id} tem horário ótimo ${optimalHour}h, agendando...`);
    // Na prática, você pode usar um job futuro ou enviar mesmo assim
    // Por ora, vamos enviar mas registrar que foi otimizado
  }

  // 2. Busca histórico de interações para personalizar
  const history = await db.select()
    .from(campaign_sent_logs)
    .where(eq(campaign_sent_logs.lead_id, lead.id))
    .orderBy(desc(campaign_sent_logs.sent_at))
    .limit(5);

  // 3. Gera mensagem personalizada via IA
  const systemPrompt = `
Você é um especialista em marketing de campanhas B2B da VendorAI.
Crie uma mensagem persuasiva para o lead abaixo, considerando:

Lead:
- Nome: ${lead.name}
- Empresa: ${lead.company || 'N/A'}
- Setor: ${lead.industry || 'N/A'}
- Estágio: ${lead.stage}
- Qualidade: ${lead.quality_score || 'N/A'}

Campanha:
- Nome: ${campaign.name}
- Descrição: ${campaign.description || 'Oferta especial'}

Histórico de envios anteriores (campanhas): ${history.length > 0 ? history.map(h => h.message_text).join('\n') : 'Nenhum'}

Regras:
- Se o lead já recebeu campanhas similares, mude a abordagem.
- Se o lead tem qualidade alta (> 80), seja mais direto (conversão).
- Se o lead tem qualidade média (50-80), seja mais consultivo (eduque).
- Se o lead tem qualidade baixa (< 50), seja mais curto (gatilho de curiosidade).

Use a ferramenta "generateMessageVariants" para criar duas variações (A e B).
`;

  const response = await openai.chat.completions.create({
    model: 'gpt-4o-mini',
    messages: [
      { role: 'system', content: systemPrompt },
      { role: 'user', content: `Crie mensagens para ${lead.name} da empresa ${lead.company || 'não informada'}` }
    ],
    tools: campaignToolSchemas.filter(t => t.function.name === 'generateMessageVariants'),
    tool_choice: { type: 'function', function: { name: 'generateMessageVariants' } },
    temperature: 0.7,
  });

  const toolCall = response.choices[0].message.tool_calls?.[0];
  if (!toolCall) {
    console.log(`[CAMPAIGN] IA não gerou mensagem para lead ${lead.id}`);
    return;
  }

  const args = JSON.parse(toolCall.function.arguments);
  console.log(`[CAMPAIGN] IA gerou variantes A e B. Recomendada: ${args.recommended_variant}`);

  // 4. Escolhe a variante (se A/B test ativo, testa amostra)
  let selectedVariant = args.recommended_variant;
  let selectedMessage = selectedVariant === 'A' ? args.variant_a : args.variant_b;

  // Se A/B test estiver ativo, envia a variante B para uma amostra
  if (campaign.ab_test_enabled) {
    const totalLeads = await db.select({ count: count() }).from(leads)
      .where(eq(leads.tenant_id, campaign.tenant_id));
    
    // Usa o ID do lead para decidir se vai para amostra (hash simples)
    const isSample = (lead.id % 100) < (campaign.ab_test_sample_size || 20);
    if (isSample) {
      selectedVariant = selectedVariant === 'A' ? 'B' : 'A';
      selectedMessage = selectedVariant === 'A' ? args.variant_a : args.variant_b;
      console.log(`[CAMPAIGN] Lead ${lead.id} selecionado para teste da variante ${selectedVariant}`);
    }
  }

  // 5. Envia via WhatsApp
  const message = `📢 *${campaign.name}*\n\n${selectedMessage}\n\n${campaign.url ? `🔗 Saiba mais: ${campaign.url}` : ''}`;

  try {
    await sendWhatsApp(lead.phone, message);
    
    // 6. Registra envio
    const log = await logCampaignSent(campaign.id, lead.id, selectedVariant, selectedMessage);
    
    // 7. Atualiza horário ótimo (após envio bem-sucedido)
    const leadInteractions = await db.select()
      .from(campaign_sent_logs)
      .where(eq(campaign_sent_logs.lead_id, lead.id));
    
    await updateOptimalHour(lead.id, leadInteractions);

    console.log(`[CAMPAIGN] Mensagem enviada para lead ${lead.id} (Variante ${selectedVariant})`);
  } catch (error) {
    console.error(`[CAMPAIGN] Erro ao enviar para lead ${lead.id}:`, error);
  }
};

// Função para webhook de interação (abertura/clique/conversão)
export const handleCampaignInteraction = async (leadId: number, campaignId: number, type: 'opened' | 'clicked' | 'converted') => {
  console.log(`[CAMPAIGN] Lead ${leadId} ${type} campanha ${campaignId}`);
  
  // Busca o log de envio mais recente para esta campanha e lead
  const [log] = await db.select()
    .from(campaign_sent_logs)
    .where(
      and(
        eq(campaign_sent_logs.campaign_id, campaignId),
        eq(campaign_sent_logs.lead_id, leadId)
      )
    )
    .orderBy(desc(campaign_sent_logs.sent_at))
    .limit(1);

  if (!log) {
    console.warn(`[CAMPAIGN] Log não encontrado para lead ${leadId}, campanha ${campaignId}`);
    return;
  }

  await logCampaignInteraction(log.id, type);
};
```

```typescript
// src/routes/campaign-webhook.ts
import express from 'express';
import { handleCampaignInteraction } from '../agents/campaign-orchestrator';

const router = express.Router();

// Endpoint para rastrear interações (use em links com tracking)
router.post('/api/campaign/track', async (req, res) => {
  try {
    const { leadId, campaignId, type } = req.body;
    
    if (!leadId || !campaignId || !['opened', 'clicked', 'converted'].includes(type)) {
      return res.status(400).json({ error: 'Dados inválidos' });
    }

    await handleCampaignInteraction(parseInt(leadId), parseInt(campaignId), type as any);
    res.json({ success: true });
  } catch (error) {
    console.error('[CAMPAIGN WEBHOOK] Erro:', error);
    res.status(500).json({ error: 'Internal error' });
  }
});

export default router;
```
