📋 MELHORIA 1: Webhooks para agentes "cegos"

Agentes afetados: Prospector, Churn, Campaigns, Revenue Scout, Partner Prospector, Recém-Abertas, SDR Follow-up, No-Response Closing.

O que faz: Adiciona endpoints para receber feedback externo (ex: confirmação de abertura, conversão, rejeição) e atualiza o sistema em tempo real.

📂 ARQUIVO: src/routes/agent-webhooks.ts

```typescript
// src/routes/agent-webhooks.ts
import express from 'express';
import { db } from '../db';
import { leads, churn_actions, campaigns, revenue_opportunities, campaign_sent_logs } from '../schema';
import { eq, and } from 'drizzle-orm';

const router = express.Router();

// 1. Webhook para Prospector: feedback sobre lead qualificado
router.post('/webhooks/prospector/feedback', async (req, res) => {
  try {
    const { leadId, feedback, rating } = req.body;
    if (!leadId) return res.status(400).json({ error: 'leadId é obrigatório' });
    
    await db.update(leads)
      .set({ 
        prospector_feedback: feedback || null,
        quality_score: rating || null 
      })
      .where(eq(leads.id, leadId));
    
    console.log(`[WEBHOOK] Prospector: lead ${leadId} recebeu feedback: ${feedback}`);
    res.json({ success: true });
  } catch (error) {
    console.error('[WEBHOOK] Erro:', error);
    res.status(500).json({ error: 'Internal error' });
  }
});

// 2. Webhook para Churn: ação de retenção foi bem-sucedida?
router.post('/webhooks/churn/action-result', async (req, res) => {
  try {
    const { actionId, successful, notes } = req.body;
    if (!actionId) return res.status(400).json({ error: 'actionId é obrigatório' });
    
    await db.update(churn_actions)
      .set({ 
        status: successful ? 'success' : 'failed',
        notes: notes || null
      })
      .where(eq(churn_actions.id, actionId));
    
    console.log(`[WEBHOOK] Churn: ação ${actionId} ${successful ? 'bem-sucedida' : 'falhou'}`);
    res.json({ success: true });
  } catch (error) {
    console.error('[WEBHOOK] Erro:', error);
    res.status(500).json({ error: 'Internal error' });
  }
});

// 3. Webhook para Campaigns: conversão de campanha (ex: lead comprou)
router.post('/webhooks/campaigns/conversion', async (req, res) => {
  try {
    const { leadId, campaignId } = req.body;
    if (!leadId || !campaignId) {
      return res.status(400).json({ error: 'leadId e campaignId são obrigatórios' });
    }
    
    await db.update(campaign_sent_logs)
      .set({ converted_at: new Date() })
      .where(
        and(
          eq(campaign_sent_logs.lead_id, leadId),
          eq(campaign_sent_logs.campaign_id, campaignId)
        )
      );
    
    // Atualiza a contagem de conversões da campanha
    await db.update(campaigns)
      .set({ converted_count: sql`${campaigns.converted_count} + 1` })
      .where(eq(campaigns.id, campaignId));
    
    console.log(`[WEBHOOK] Campaigns: lead ${leadId} converteu na campanha ${campaignId}`);
    res.json({ success: true });
  } catch (error) {
    console.error('[WEBHOOK] Erro:', error);
    res.status(500).json({ error: 'Internal error' });
  }
});

// 4. Webhook para Revenue Scout: oportunidade ganha/perdida
router.post('/webhooks/revenue/opportunity', async (req, res) => {
  try {
    const { opportunityId, status, notes } = req.body; // 'won' ou 'lost'
    if (!opportunityId || !['won', 'lost'].includes(status)) {
      return res.status(400).json({ error: 'opportunityId e status (won/lost) são obrigatórios' });
    }
    
    await db.update(revenue_opportunities)
      .set({ 
        stage: status === 'won' ? 'won' : 'lost',
        updated_at: new Date(),
        notes: notes || null
      })
      .where(eq(revenue_opportunities.id, opportunityId));
    
    console.log(`[WEBHOOK] Revenue: oportunidade ${opportunityId} ${status}`);
    res.json({ success: true });
  } catch (error) {
    console.error('[WEBHOOK] Erro:', error);
    res.status(500).json({ error: 'Internal error' });
  }
});

// 5. Webhook para Abertura de Mensagem (read receipt da Meta)
router.post('/webhooks/meta/read-receipt', async (req, res) => {
  try {
    const { wamid, status } = req.body;
    if (!wamid) return res.status(400).json({ error: 'wamid é obrigatório' });
    
    // Atualiza o status da mensagem no banco
    // Usar wamid para buscar o message_id e atualizar opened_at
    // Assumindo que existe uma tabela messages com campo wamid
    await db.update(messages)
      .set({ 
        read_at: new Date(),
        status: status || 'delivered'
      })
      .where(eq(messages.wamid, wamid));
    
    console.log(`[WEBHOOK] Meta read-receipt: ${wamid} lida`);
    res.json({ success: true });
  } catch (error) {
    console.error('[WEBHOOK] Erro:', error);
    res.status(500).json({ error: 'Internal error' });
  }
});

// 6. Webhook para Campaigns: rastreamento de clique
router.post('/webhooks/campaigns/click', async (req, res) => {
  try {
    const { leadId, campaignId, url } = req.body;
    if (!leadId || !campaignId) {
      return res.status(400).json({ error: 'leadId e campaignId são obrigatórios' });
    }
    
    // Busca o log de envio mais recente
    const [log] = await db.select()
      .from(campaign_sent_logs)
      .where(
        and(
          eq(campaign_sent_logs.lead_id, leadId),
          eq(campaign_sent_logs.campaign_id, campaignId)
        )
      )
      .orderBy(campaign_sent_logs.sent_at, 'desc')
      .limit(1);
    
    if (log) {
      await db.update(campaign_sent_logs)
        .set({ 
          clicked_at: new Date(),
          clicked_url: url || null
        })
        .where(eq(campaign_sent_logs.id, log.id));
      
      // Atualiza contagem de cliques da campanha
      await db.update(campaigns)
        .set({ clicked_count: sql`${campaigns.clicked_count} + 1` })
        .where(eq(campaigns.id, campaignId));
    }
    
    console.log(`[WEBHOOK] Campaigns: lead ${leadId} clicou na campanha ${campaignId}`);
    res.json({ success: true });
  } catch (error) {
    console.error('[WEBHOOK] Erro:', error);
    res.status(500).json({ error: 'Internal error' });
  }
});

// 7. Webhook para Churn: registrar churn real (via integração externa)
router.post('/webhooks/churn/register', async (req, res) => {
  try {
    const { leadId, userId, reason } = req.body;
    if (!leadId) return res.status(400).json({ error: 'leadId é obrigatório' });
    
    await db.update(leads)
      .set({ 
        is_churned: true,
        churned_at: new Date(),
        churn_reason: reason || null,
        stage: 'Perdido'
      })
      .where(eq(leads.id, leadId));
    
    // Atualiza o histórico de churn
    const [history] = await db.select()
      .from(churn_history)
      .where(
        and(
          eq(churn_history.lead_id, leadId),
          isNull(churn_history.churned_at)
        )
      )
      .orderBy(churn_history.predicted_at, 'desc')
      .limit(1);
    
    if (history) {
      await db.update(churn_history)
        .set({ churned_at: new Date() })
        .where(eq(churn_history.id, history.id));
    }
    
    console.log(`[WEBHOOK] Churn: lead ${leadId} registrado como churn`);
    res.json({ success: true });
  } catch (error) {
    console.error('[WEBHOOK] Erro:', error);
    res.status(500).json({ error: 'Internal error' });
  }
});

export default router;
```

    return;
  }

  // Busca dados do lead
  const [lead] = await db.select().from(leads).where(eq(leads.id, leadId));
  if (!lead) {
    console.warn(`[CRM] Lead ${leadId} não encontrado.`);
    return;
  }

  const payload = {
    event,
    lead: {
      id: lead.id,
      name: lead.name,
      company: lead.company,
      phone: lead.phone,
      email: lead.email,
      stage: lead.stage,
      quality_score: lead.quality_score,
      total_revenue: lead.total_revenue,
      last_purchase_at: lead.last_purchase_at,
      source: lead.source || 'vendorai',
      created_at: lead.created_at,
    },
    timestamp: new Date().toISOString(),
  };

  try {
    const headers: any = {
      'Content-Type': 'application/json',
    };
    if (CRM_API_KEY) {
      headers['Authorization'] = `Bearer ${CRM_API_KEY}`;
    }

    await axios.post(CRM_WEBHOOK_URL, payload, { headers });
    console.log(`[CRM] Lead ${leadId} exportado para CRM (evento: ${event})`);
  } catch (error: any) {
    console.error('[CRM] Erro ao exportar:', error.response?.data || error.message);
  }
};

// Função para exportar em lote
export const exportBatchToCRM = async (leadIds: number[], event: 'qualified' | 'client' | 'lead') => {
  console.log(`[CRM] Exportando ${leadIds.length} leads para CRM...`);
  for (const id of leadIds) {
    await exportLeadToCRM(id, event);
  }

// src/agents/cache.ts
import { createClient } from 'redis';

let redisClient: any = null;

// Inicializar Redis (Upstash)
export const initCache = async () => {
  const redisUrl = process.env.UPSTASH_REDIS_URL;
  if (!redisUrl) {
    console.warn('[CACHE] UPSTASH_REDIS_URL não configurado. Usando cache em memória fallback.');
    return;
  }
  
  try {
    redisClient = createClient({ url: redisUrl });
    await redisClient.connect();
    console.log('[CACHE] Redis conectado.');
  } catch (error) {
    console.warn('[CACHE] Redis não disponível. Usando cache em memória fallback.');
    redisClient = null;
  }
};

// Cache em memória (fallback)
const memoryCache = new Map<string, { data: any; expires: number }>();

// Obter resposta em cache
export const getCachedResponse = async (key: string): Promise<any | null> => {
  // Tenta Redis primeiro
  if (redisClient) {
    try {
      const cached = await redisClient.get(`cache:${key}`);
      if (cached) {
        const parsed = JSON.parse(cached);
        if (parsed.expires > Date.now()) {
          return parsed.data;
        }
      }
    } catch (error) {
      console.warn('[CACHE] Erro no Redis, usando fallback:', error);
    }
  }
  
  // Fallback: memória
  const cached = memoryCache.get(key);
  if (cached && cached.expires > Date.now()) {
    return cached.data;
  }
  
  return null;
};

// Armazenar resposta em cache
export const cacheResponse = async (
  key: string, 
  data: any, 
  ttlSeconds: number = 3600 // 1 hora padrão
): Promise<void> => {
  const expires = Date.now() + (ttlSeconds * 1000);
  const payload = { data, expires };
  
  // Tenta Redis primeiro
  if (redisClient) {
    try {
      await redisClient.set(`cache:${key}`, JSON.stringify(payload), { EX: ttlSeconds });
      return;
    } catch (error) {
      console.warn('[CACHE] Erro no Redis, usando fallback:', error);
    }
  }
  
  // Fallback: memória
  memoryCache.set(key, payload);
  
  // Limpar após expirar (opcional)
  setTimeout(() => {
    memoryCache.delete(key);
  }, ttlSeconds * 1000);
};

// Limpar cache de um agente específico
export const clearAgentCache = async (agentName: string) => {
  const pattern = `cache:${agentName}:*`;
  if (redisClient) {
    try {
      const keys = await redisClient.keys(pattern);
      if (keys.length > 0) {
        await redisClient.del(keys);
      }
    } catch (error) {
      console.warn('[CACHE] Erro ao limpar cache:', error);
    }
  }
  
  // Limpar memória
  for (const key of memoryCache.keys()) {
    if (key.startsWith(agentName)) {
      memoryCache.delete(key);
    }
  }
};

// Gerar chave de cache (baseada no prompt + entrada)
export const generateCacheKey = (agentName: string, input: any): string => {
  const normalized = JSON.stringify(input).replace(/\s/g, '');
  return `${agentName}:${normalized.substring(0, 100)}`;
};

// src/agents/feedback-loop.ts
import { db } from '../db';
import { prospect_feedback, churn_actions, leads, prospector_config } from '../schema';
import { eq, and, gte, lte, sql, desc } from 'drizzle-orm';

// 1. Ajustar pesos do Prospector com base em feedback
export const adjustProspectorWeights = async (tenantId: string) => {
  console.log('[FEEDBACK] Ajustando pesos do Prospector...');

  // Busca leads com feedback (rating)
  const feedbacks = await db.select({
    lead_id: prospect_feedback.lead_id,
    rating: prospect_feedback.rating,
    converted: prospect_feedback.converted,
  })
  .from(prospect_feedback)
  .innerJoin(leads, eq(leads.id, prospect_feedback.lead_id))
  .where(eq(leads.tenant_id, tenantId));

  if (feedbacks.length < 10) {
    console.log('[FEEDBACK] Dados insuficientes (menos de 10 feedbacks).');
    return;
  }

  // Analisa leads bons (rating >= 4 e convertidos) vs ruins (rating <= 2)
  const goodLeads = feedbacks.filter(f => f.rating >= 4 && f.converted);
  const badLeads = feedbacks.filter(f => f.rating <= 2);

  if (goodLeads.length === 0 || badLeads.length === 0) {
    console.log('[FEEDBACK] Falta dados de comparação (bons ou ruins).');
    return;
  }

  // Busca dados enriquecidos dos leads
  const goodLeadIds = goodLeads.map(l => l.lead_id);
  const badLeadIds = badLeads.map(l => l.lead_id);

  const goodData = await db.select().from(leads).where(inArray(leads.id, goodLeadIds));
  const badData = await db.select().from(leads).where(inArray(leads.id, badLeadIds));

  // Calcula proporção de cada característica
  const goodHasWebsite = goodData.filter(l => l.website).length / goodData.length;
  const badHasWebsite = badData.filter(l => l.website).length / badData.length;
  
  const goodHasPhone = goodData.filter(l => l.phone).length / goodData.length;
  const badHasPhone = badData.filter(l => l.phone).length / badData.length;

  const goodHasAds = goodData.filter(l => l.has_ads).length / goodData.length;
  const badHasAds = badData.filter(l => l.has_ads).length / badData.length;

  // Ajusta pesos baseado na diferença
  const weights = {
    has_website: Math.round((goodHasWebsite / (badHasWebsite || 0.1)) * 20),
    has_phone: Math.round((goodHasPhone / (badHasPhone || 0.1)) * 15),
    has_ads: Math.round((goodHasAds / (badHasAds || 0.1)) * 25),
    rating_google: 20,
    recent_opening: 15,
    has_crm: 15,
  };

  // Salva novos pesos na configuração
  await db.update(prospector_config)
    .set({
      weights: weights,
      updated_at: new Date(),
    })
    .where(eq(prospector_config.tenant_id, tenantId));

  console.log('[FEEDBACK] Pesos ajustados:', weights);
};

// 2. Ajustar thresholds do Churn com base em histórico
export const adjustChurnThresholds = async (tenantId: string) => {
  console.log('[FEEDBACK] Ajustando thresholds do Churn...');

  // Busca ações de retenção bem-sucedidas vs malsucedidas
  const successfulActions = await db.select()
    .from(churn_actions)
    .where(
      and(
        eq(churn_actions.status, 'success'),
        sql`${churn_actions.executed_at} > NOW() - INTERVAL '30 days'`
      )
    );

  const failedActions = await db.select()
    .from(churn_actions)
    .where(
      and(
        eq(churn_actions.status, 'failed'),
        sql`${churn_actions.executed_at} > NOW() - INTERVAL '30 days'`
      )
    );

  if (successfulActions.length === 0 || failedActions.length === 0) {
    console.log('[FEEDBACK] Dados insuficientes para ajustar thresholds.');
    return;
  }

  // Análise: quais ações funcionam melhor?
  const actionTypes = ['discount_offer', 'content_sent', 'escalated_sdr', 'call_scheduled'];
  const successRates: Record<string, number> = {};

  for (const type of actionTypes) {
    const total = successfulActions.filter(a => a.action_type === type).length;
    const failed = failedActions.filter(a => a.action_type === type).length;
    const rate = total + failed > 0 ? total / (total + failed) : 0;
    successRates[type] = rate;
  }

  // Sugere novos thresholds baseado na taxa de sucesso
  const thresholds = {
    critical: 80, // mantém
    high: successRates.discount_offer > 0.5 ? 60 : 70,
    medium: successRates.content_sent > 0.5 ? 50 : 60,
    low: 40,
  };

  // Salva novos thresholds (em uma tabela de configuração, se existir)
  // Ou retorna para uso manual

  console.log('[FEEDBACK] Novos thresholds sugeridos:', thresholds);
  return thresholds;
};

// 3. Ajustar pontuação do Revenue Scout
export const adjustRevenueScoring = async (tenantId: string) => {
  console.log('[FEEDBACK] Ajustando scoring do Revenue Scout...');

  // Busca oportunidades ganhas vs perdidas
  const wonOpportunities = await db.select()
    .from(revenue_opportunities)
    .where(
      and(
        eq(revenue_opportunities.tenant_id, tenantId),
        eq(revenue_opportunities.stage, 'won')
      )
    );

  const lostOpportunities = await db.select()
    .from(revenue_opportunities)
    .where(
      and(
        eq(revenue_opportunities.tenant_id, tenantId),
        eq(revenue_opportunities.stage, 'lost')
      )
    );

  if (wonOpportunities.length === 0 || lostOpportunities.length === 0) {
    console.log('[FEEDBACK] Dados insuficientes para ajustar scoring.');
    return;
  }

  // Analisa scores médios
  const avgWonScore = wonOpportunities.reduce((sum, o) => sum + (o.opportunity_score || 0), 0) / wonOpportunities.length;
  const avgLostScore = lostOpportunities.reduce((sum, o) => sum + (o.opportunity_score || 0), 0) / lostOpportunities.length;

  // Ajusta threshold de ação
  const newThreshold = Math.round((avgWonScore + avgLostScore) / 2);

  // Salva novo threshold (em prospector_config ou revenue_config)
  await db.update(prospector_config)
    .set({
      auto_dispatch_threshold: newThreshold,
      updated_at: new Date(),
    })
    .where(eq(prospector_config.tenant_id, tenantId));

  console.log(`[FEEDBACK] Novo threshold do Revenue Scout: ${newThreshold}`);
};

  await adjustChurnThresholds(tenantId);
  await adjustRevenueScoring(tenantId);
